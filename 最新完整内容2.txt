def zhang_suen_thinning(src):
    """输入 uint8 二值图，前景非零；输出 0/255 骨架。"""
    im = (src > 0).astype(np.uint8)
    changed = True
    while changed:
        changed = False
        for phase in (0, 1):
            p = np.pad(im, 1, mode="constant")
            P2 = p[:-2, 1:-1]
            P3 = p[:-2, 2:]
            P4 = p[1:-1, 2:]
            P5 = p[2:, 2:]
            P6 = p[2:, 1:-1]
            P7 = p[2:, :-2]
            P8 = p[1:-1, :-2]
            P9 = p[:-2, :-2]
            P1 = im
            neighbors = P2 + P3 + P4 + P5 + P6 + P7 + P8 + P9
            transitions = (
                ((P2 == 0) & (P3 == 1)).astype(np.uint8) +
                ((P3 == 0) & (P4 == 1)).astype(np.uint8) +
                ((P4 == 0) & (P5 == 1)).astype(np.uint8) +
                ((P5 == 0) & (P6 == 1)).astype(np.uint8) +
                ((P6 == 0) & (P7 == 1)).astype(np.uint8) +
                ((P7 == 0) & (P8 == 1)).astype(np.uint8) +
                ((P8 == 0) & (P9 == 1)).astype(np.uint8) +
                ((P9 == 0) & (P2 == 1)).astype(np.uint8)
            )
            if phase == 0:
                c1 = P2 * P4 * P6
                c2 = P4 * P6 * P8
            else:
                c1 = P2 * P4 * P8
                c2 = P2 * P6 * P8
            remove = (
                (P1 == 1) &
                (neighbors >= 2) & (neighbors <= 6) &
                (transitions == 1) &
                (c1 == 0) & (c2 == 0)
            )
            if np.any(remove):
                im[remove] = 0
                changed = True
    return (im * 255).astype(np.uint8)

import hou
import math
from collections import defaultdict

# ============================================================
# 参数
# ============================================================
ENDPOINT_TOL = 0.08
MIN_ANGLE_DEG = 150
AMBIGUOUS_ANGLE_MARGIN = 5.0
ANGLE_TIE_EPSILON = 1.0
DISTANCE_TIE_EPSILON = 1e-6
# ============================================================

node = hou.pwd()
inputs = node.inputs()
if not inputs:
    raise hou.NodeError("python_topology_merge_v16 需要连接输入曲线。")

src = inputs[0].geometry()
geo = node.geometry()

# 读取输入
curves = []
closed_count = 0
unsupported_count = 0
for prim in src.prims():
    try:
        vertices = list(prim.vertices())
        points = [v.point().position() for v in vertices]
        is_closed = bool(prim.isClosed())
    except Exception:
        unsupported_count += 1
        continue
    if len(points) < 2:
        continue
    curves.append({
        "id": int(prim.number()),
        "points": points,
        "closed": is_closed,
    })
    if is_closed:
        closed_count += 1

n = len(curves)
if n == 0:
    raise hou.NodeError("输入中没有可处理的曲线 Primitive。")

# -------------------- 端点信息 --------------------
endpoints = {}
open_endpoint_keys = []
for ci, curve in enumerate(curves):
    if curve["closed"]:
        continue
    pts = curve["points"]
    start_vec = pts[1] - pts[0]
    end_vec = pts[-2] - pts[-1]
    for side, pos, vec in (
        (0, pts[0], start_vec),
        (1, pts[-1], end_vec),
    ):
        key = (ci, side)
        length = vec.length()
        if length > 1e-12:
            direction = vec / length
        else:
            direction = None
        endpoints[key] = {
            "pos": pos,
            "direction": direction,
            "curve_index": ci,
            "side": side,
        }
        open_endpoint_keys.append(key)

# -------------------- 候选端点对 --------------------
def angle_between_degrees(a, b):
    if a is None or b is None:
        return None
    dot = max(-1.0, min(1.0, a.dot(b)))
    return math.degrees(math.acos(dot))

candidates = []
by_endpoint = defaultdict(list)
keys = list(endpoints.keys())
# V16：用 XZ 空间哈希查找端点近邻，取代所有端点的 O(N^2) 两两比较。
# 网格只用来生成候选，最终仍以 3D 距离和切线夹角精确判断。
endpoint_cell = max(float(ENDPOINT_TOL), 1e-6)
endpoint_grid = defaultdict(list)
for i, ka in enumerate(keys):
    ea = endpoints[ka]
    gx = int(math.floor(float(ea["pos"][0]) / endpoint_cell))
    gz = int(math.floor(float(ea["pos"][2]) / endpoint_cell))
    for dx in (-1, 0, 1):
        for dz in (-1, 0, 1):
            for j in endpoint_grid.get((gx + dx, gz + dz), []):
                kb = keys[j]
                eb = endpoints[kb]
                if ea["curve_index"] == eb["curve_index"]:
                    continue
                distance = (ea["pos"] - eb["pos"]).length()
                if distance > ENDPOINT_TOL:
                    continue
                angle = angle_between_degrees(ea["direction"], eb["direction"])
                if angle is None or angle < MIN_ANGLE_DEG:
                    continue
                candidate = {"a": ka, "b": kb, "distance": distance, "angle": angle}
                candidates.append(candidate)
                by_endpoint[ka].append(candidate)
                by_endpoint[kb].append(candidate)
    endpoint_grid[(gx, gz)].append(i)

def other_endpoint(candidate, key):
    return candidate["b"] if candidate["a"] == key else candidate["a"]

def candidate_rank(candidate):
    return (-candidate["angle"], candidate["distance"])

# -------------------- 最佳候选与歧义判定 --------------------
best_for_endpoint = {}
ambiguous_endpoints = set()
for key, options in by_endpoint.items():
    options.sort(key=candidate_rank)
    best = options[0]
    best_for_endpoint[key] = best
    if len(options) >= 2:
        second = options[1]
        angle_gap = best["angle"] - second["angle"]
        distance_gap = abs(best["distance"] - second["distance"])
        near_tie = (
            angle_gap <= ANGLE_TIE_EPSILON
            and distance_gap <= DISTANCE_TIE_EPSILON
        )
        weak_margin = angle_gap < AMBIGUOUS_ANGLE_MARGIN
        if near_tie or weak_margin:
            ambiguous_endpoints.add(key)

# -------------------- 互为最佳匹配才接受 --------------------
accepted = []
rejected_ambiguous = 0
for candidate in candidates:
    a = candidate["a"]
    b = candidate["b"]
    if a in ambiguous_endpoints or b in ambiguous_endpoints:
        rejected_ambiguous += 1
        continue
    best_a = best_for_endpoint.get(a)
    best_b = best_for_endpoint.get(b)
    if best_a is candidate and best_b is candidate:
        accepted.append(candidate)

connections = {}
for candidate in accepted:
    a = candidate["a"]
    b = candidate["b"]
    connections[a] = b
    connections[b] = a

# -------------------- 重建连续曲线 --------------------
visited = set()
output_paths = []

def points_close(a, b, tolerance):
    return (a - b).length() <= tolerance

def oriented_points(curve_index, reverse):
    pts = curves[curve_index]["points"]
    return list(reversed(pts)) if reverse else list(pts)

def build_chain(start_curve, start_side, cycle_mode=False):
    path = []
    current_curve = start_curve
    reverse = (start_side == 1)
    first_curve = start_curve
    closed_chain = False
    while True:
        if current_curve in visited:
            if cycle_mode and current_curve == first_curve:
                closed_chain = True
            break
        visited.add(current_curve)
        pts = oriented_points(current_curve, reverse)
        if not path:
            path.extend(pts)
        else:
            if points_close(path[-1], pts[0], ENDPOINT_TOL):
                path.extend(pts[1:])
            else:
                midpoint = (path[-1] + pts[0]) * 0.5
                path[-1] = midpoint
                pts[0] = midpoint
                path.extend(pts[1:])
        exit_side = 0 if reverse else 1
        exit_key = (current_curve, exit_side)
        next_key = connections.get(exit_key)
        if next_key is None:
            break
        next_curve, next_side = next_key
        if next_curve == first_curve:
            closed_chain = True
            break
        if next_curve in visited:
            break
        current_curve = next_curve
        reverse = (next_side == 1)
    if closed_chain and len(path) >= 3:
        if points_close(path[0], path[-1], ENDPOINT_TOL):
            path.pop()
    return path, closed_chain

# 第一遍：从开放端点开始构建
for ci, curve in enumerate(curves):
    if curve["closed"] or ci in visited:
        continue
    start_key = (ci, 0)
    end_key = (ci, 1)
    start_connected = start_key in connections
    end_connected = end_key in connections
    if not start_connected:
        path, is_closed = build_chain(ci, 0)
        if len(path) >= 2:
            output_paths.append((path, is_closed))
    elif not end_connected:
        path, is_closed = build_chain(ci, 1)
        if len(path) >= 2:
            output_paths.append((path, is_closed))

# 第二遍：剩余闭环组件
for ci, curve in enumerate(curves):
    if curve["closed"] or ci in visited:
        continue
    path, is_closed = build_chain(ci, 0, cycle_mode=True)
    if len(path) >= 2:
        output_paths.append((path, is_closed))

# 原有闭合曲线直接保留
for curve in curves:
    if curve["closed"]:
        output_paths.append((curve["points"], True))

# -------------------- 输出几何 --------------------
geo.clear()
for pts, is_closed in output_paths:
    if len(pts) < 2:
        continue
    polygon = geo.createPolygon()
    polygon.setIsClosed(bool(is_closed))
    for pos in pts:
        point = geo.createPoint()
        point.setPosition(pos)
        polygon.addVertex(point)

# -------------------- 统计信息 --------------------
original_open_count = sum(
    1 for curve in curves if not curve["closed"]
)
merged_count = original_open_count - sum(
    1 for _, is_closed in output_paths if not is_closed
)

def path_length(points, is_closed):
    if len(points) < 2:
        return 0.0
    length = 0.0
    for i in range(1, len(points)):
        length += (points[i] - points[i - 1]).length()
    if is_closed:
        length += (points[0] - points[-1]).length()
    return length

length_before = sum(
    path_length(c["points"], c["closed"]) for c in curves
)
length_after = sum(
    path_length(pts, closed) for pts, closed in output_paths
)

report = """
========== python_topology_merge_v16 ==========
Input curve primitives: {}
Input open curves: {}
Input closed curves: {}
Input endpoints: {}
Candidate endpoint pairs: {}
Ambiguous endpoints: {}
Candidates rejected due to ambiguity: {}
Accepted connections: {}
Output curve primitives: {}
Output closed curves: {}
Total length before: {:.6f}
Total length after: {:.6f}
Unsupported primitives skipped: {}
============================================
""".format(
    len(curves),
    original_open_count,
    closed_count,
    len(open_endpoint_keys),
    len(candidates),
    len(ambiguous_endpoints),
    rejected_ambiguous,
    len(accepted),
    len(output_paths),
    sum(1 for _, closed in output_paths if closed),
    length_before,
    length_after,
    unsupported_count,
)
print(report)
print("V16 endpoint spatial-hash candidate generation enabled; endpoint_cell={:.5f}; min_angle={:.1f}".format(endpoint_cell, MIN_ANGLE_DEG))

geo.addAttrib(hou.attribType.Global, "topology_merge_report", "")
geo.setGlobalAttribValue("topology_merge_report", report)

# -*- coding: utf-8 -*-
"""Houdini 20.5 Python SOP — V16 anchor-constrained polyline smoothing.
Input: cleaned/merged polygon curves. Output: same primitive topology with conservatively smoothed point positions.
Important: this is constrained polyline smoothing, not vector/Bézier reconstruction.
"""
import hou
import math
import time
from collections import defaultdict

# -------------------- User parameters --------------------
SMOOTH_LAMBDA = 0.18             # 每轮向相邻顶点中点移动的比例，0~1。
SMOOTH_ITERATIONS = 4            # 平滑迭代次数；过大可能损失书法细节。
HARD_CORNER_TURN_DEG = 52.0      # 跨窗口转向角超过该值才作为硬锚点，降低像素级锯齿被误认作真拐角的概率。
CORNER_WINDOW_RADIUS = 4          # 使用前后多少个顶点估计转向；图像骨架像素很密时可设 3~6。
ANCHOR_POSITION_TOL = 0.025      # 不同 Primitive 顶点在该距离内重合时，作为几何共享锚点。
MAX_TOTAL_DISPLACEMENT = 0.12    # 任一点相对原坐标的最大累计位移。
PROTECT_OPEN_ENDPOINTS = True   # 固定所有开放曲线首尾端点。
PROTECT_SHARED_GEOMETRY = True  # 固定不同曲线中空间重合的顶点，即便 Point number 不同。
PROTECT_CORNER_VERTICES = True  # 固定明显转角，保护文字角部和短钩。
MEDIAN_PASSES = 0                # 可选轻量中值预平滑次数；默认 0 以保护小笔画。
EPS = 1e-9
# --------------------------------------------------------
start = time.perf_counter()
node = hou.pwd()
if not node.inputs():
    raise hou.NodeError("python_curve_smooth_v16：请连接 python_topology_merge_v16。")
src = node.inputs()[0].geometry()
geo = node.geometry(); geo.clear(); geo.merge(src)
prims = list(geo.prims()); points = list(geo.points())
if not points or not prims:
    raise hou.NodeError("python_curve_smooth_v16：输入没有有效曲线点。")

orig = {p.number(): hou.Vector3(p.position()) for p in points}
cur = dict(orig)
curves=[]
point_uses=defaultdict(set)
for prim in prims:
    seq=[]
    for v in prim.vertices():
        pid=v.point().number()
        if not seq or seq[-1] != pid: seq.append(pid)
    if len(seq)<2: continue
    closed=bool(prim.isClosed())
    curves.append({'prim':prim.number(),'seq':seq,'closed':closed})
    for pid in set(seq): point_uses[pid].add(prim.number())

def turn_angle(prev, this, nxt):
    a=this-prev; b=nxt-this
    if a.length()<EPS or b.length()<EPS: return 0.0
    return math.degrees(math.acos(max(-1.0,min(1.0,a.normalized().dot(b.normalized())))))

def window_turn(seq, i, closed, radius):
    """用较长的前后弦估计转向角，降低像素路径的逐点锯齿影响。"""
    n=len(seq)
    if n<3: return 0.0
    r=max(1,int(radius))
    if closed:
        left=seq[(i-r)%n]; center=seq[i]; right=seq[(i+r)%n]
    else:
        li=max(0,i-r); ri=min(n-1,i+r)
        if li==i or ri==i: return 0.0
        left=seq[li]; center=seq[i]; right=seq[ri]
    a=center-left; b=right-center
    if a.length()<EPS or b.length()<EPS: return 0.0
    return math.degrees(math.acos(max(-1.0,min(1.0,a.normalized().dot(b.normalized())))))

# Spatial hash for duplicate geometric vertices, not all-pairs search.
shared_geom=set()
if PROTECT_SHARED_GEOMETRY and ANCHOR_POSITION_TOL>0:
    cell=max(ANCHOR_POSITION_TOL,1e-6); buckets=defaultdict(list)
    for pid,pos in orig.items():
        key=tuple(int(math.floor(float(pos[i])/cell)) for i in range(3))
        for dx in (-1,0,1):
            for dy in (-1,0,1):
                for dz in (-1,0,1):
                    for other in buckets.get((key[0]+dx,key[1]+dy,key[2]+dz),[]):
                        if (pos-orig[other]).length()<=ANCHOR_POSITION_TOL:
                            if point_uses.get(pid,set()) != point_uses.get(other,set()) or pid!=other:
                                shared_geom.add(pid); shared_geom.add(other)
        buckets[key].append(pid)

fixed=set()
for c in curves:
    seq=c['seq']; n=len(seq); closed=c['closed']
    if PROTECT_OPEN_ENDPOINTS and not closed:
        fixed.add(seq[0]); fixed.add(seq[-1])
    for i,pid in enumerate(seq):
        if PROTECT_SHARED_GEOMETRY and (pid in shared_geom or len(point_uses.get(pid,()))>1): fixed.add(pid)
        if PROTECT_CORNER_VERTICES and (closed or 0<i<n-1):
            turn=window_turn([orig[q] for q in seq],i,closed,CORNER_WINDOW_RADIUS)
            if turn>=HARD_CORNER_TURN_DEG: fixed.add(pid)

# Optional median-like pass (very conservative and disabled by default).
for _ in range(max(0,int(MEDIAN_PASSES))):
    old=dict(cur); props=defaultdict(list)
    for c in curves:
        seq=c['seq']; n=len(seq); closed=c['closed']
        indices=range(n) if closed else range(1,n-1)
        for i in indices:
            pid=seq[i]
            if pid in fixed: continue
            prev=seq[(i-1)%n]; nxt=seq[(i+1)%n]
            props[pid].append((old[prev]+old[nxt])*0.5)
    for pid,vals in props.items():
        cur[pid]=sum(vals,hou.Vector3(0,0,0))/float(len(vals))

# Anchor-constrained Laplacian smoothing with a hard displacement cap.
for _ in range(max(0,int(SMOOTH_ITERATIONS))):
    old=dict(cur); proposals=defaultdict(list)
    for c in curves:
        seq=c['seq']; n=len(seq); closed=c['closed']
        indices=range(n) if closed else range(1,n-1)
        for i in indices:
            pid=seq[i]
            if pid in fixed: continue
            prev=seq[(i-1)%n]; nxt=seq[(i+1)%n]
            midpoint=(old[prev]+old[nxt])*0.5
            proposals[pid].append(old[pid]+(midpoint-old[pid])*float(SMOOTH_LAMBDA))
    for pid,vals in proposals.items():
        proposal=sum(vals,hou.Vector3(0,0,0))/float(len(vals))
        delta=proposal-orig[pid]; d=delta.length()
        if d>MAX_TOTAL_DISPLACEMENT and d>EPS: proposal=orig[pid]+delta*(MAX_TOTAL_DISPLACEMENT/d)
        cur[pid]=proposal
for p in geo.points(): p.setPosition(cur[p.number()])

# Detail diagnostics.
def set_detail(name,value):
    attrib=geo.findGlobalAttrib(name)
    if attrib is None: geo.addAttrib(hou.attribType.Global,name,value)
    geo.setGlobalAttribValue(name,value)
max_move=max([(cur[k]-orig[k]).length() for k in cur] or [0.0])
# Canonical V16 diagnostic names.
set_detail('smooth_v16_fixed_anchor_point_count',int(len(fixed)))
set_detail('smooth_v16_shared_geometry_anchor_count',int(len(shared_geom)))
set_detail('smooth_v16_iterations',int(SMOOTH_ITERATIONS))
set_detail('smooth_v16_corner_window_radius',int(CORNER_WINDOW_RADIUS))
set_detail('smooth_v16_max_total_displacement',float(MAX_TOTAL_DISPLACEMENT))
set_detail('smooth_v16_is_bezier_fit',0)
set_detail('smooth_v16_measured_max_displacement',float(max_move))
# Legacy aliases keep existing downstream dashboards readable.
set_detail('smooth_v15_fixed_anchor_point_count',int(len(fixed)))
set_detail('smooth_v15_shared_geometry_anchor_count',int(len(shared_geom)))
set_detail('smooth_v15_iterations',int(SMOOTH_ITERATIONS))
set_detail('smooth_v15_corner_window_radius',int(CORNER_WINDOW_RADIUS))
set_detail('smooth_v15_max_total_displacement',float(MAX_TOTAL_DISPLACEMENT))
set_detail('smooth_v15_is_bezier_fit',0)
set_detail('smooth_v15_measured_max_displacement',float(max_move))
print('python_curve_smooth_v16 | points={} curves={} fixed_anchors={} shared_geometry={} corner_window={} max_move={:.5f} cook={:.3f}s'.format(len(points),len(curves),len(fixed),len(shared_geom),CORNER_WINDOW_RADIUS,max_move,time.perf_counter()-start))

# ORIGIN：以世界原点 (0,0,0) 为中心。
# CUSTOM：使用 CUSTOM_PIVOT_X/Y/Z 指定的自定义中心。

CUSTOM_PIVOT_Y = 0.0
CUSTOM_PIVOT_Z = 0.0
# CENTER_MODE="CUSTOM" 时使用的拉伸中心坐标。

DIRECTION_ANGLE_DEG = 0.0
# 拉伸主方向在 XZ 平面中的角度，角度从 +X 朝 +Z 测量。
# 0°：水平 X 方向；90°：竖直 Z 方向；45°：斜向；-45°：反向斜向。

SCALE_ALONG = 1.0
# 沿 DIRECTION_ANGLE_DEG 指定方向的比例。>1 拉长，0~1 压缩，1 不变。

SCALE_CROSS = 1.5
# 沿主方向垂直方向（仍位于 XZ 平面内）的比例。>1 增宽，0~1 压窄，1 不变。
# 例如主方向 0° 时，SCALE_ALONG 拉伸 X，SCALE_CROSS 拉伸 Z。

SCALE_Y = 1.0
# 独立控制 Y 轴缩放比例。图片骨架通常位于 XZ 平面，Y 方向通常保持 1.0。

UNIFORM_SCALE = 1.0
# 全局附加等比例缩放，会乘到 SCALE_ALONG / SCALE_CROSS / SCALE_Y 上。
# 如果需要最终尺寸整体放大 4/3，可将此值设为 1.333333；必须在布点之前完成。

REPORT_BOUNDS = True
# True 时在 Python Shell 输出拉伸前后的 X/Y/Z 包围盒尺寸，便于检查变形效果。

def bbox_center(pos_list):
    return hou.Vector3(
        (min(p[0] for p in pos_list) + max(p[0] for p in pos_list)) * 0.5,
        (min(p[1] for p in pos_list) + max(p[1] for p in pos_list)) * 0.5,
        (min(p[2] for p in pos_list) + max(p[2] for p in pos_list)) * 0.5
    )

def bbox_size(pos_list):
    return (
        max(p[0] for p in pos_list) - min(p[0] for p in pos_list),
        max(p[1] for p in pos_list) - min(p[1] for p in pos_list),
        max(p[2] for p in pos_list) - min(p[2] for p in pos_list)
    )

    center = hou.Vector3((0.0, 0.0, 0.0))
elif CENTER_MODE.upper() == "CUSTOM":

    print("Bounds before         :", tuple(round(vv, 6) for vv in bbox_size(before)))
    print("Bounds after          :", tuple(round(vv, 6) for vv in bbox_size(after)))
    print("===============================================")

import hou
import math
import time
from collections import defaultdict

# ============================================================================
# python_stroke_analysis_v16 — Houdini 20.5 Python SOP
# 功能：不改变曲线位置，为每条 Primitive 估计几何类型、主方向、曲率特征，
#       并按交接/相近关系建立几何关联组。不是 OCR，不保证推断汉字语义笔顺；shape_group_id 仅是几何邻近关联，不等于语义笔画链。
# ============================================================================

# ---- 几何类型分类参数 ----
SHORT_FEATURE_MAX_LENGTH = 0.75
# 总弧长小于该值且不是闭环时，标记为 short_feature。它仅是几何长度标签，不代表汉字语义中的“点”。单位为输入几何世界单位。

CURVE_PATH_CHORD_RATIO = 1.06
# 开放曲线的弧长/端点弦长超过该值时，倾向标记 curved 或 arc。
# 波浪“一”如果比较平、弦长很长，可把该值降到 1.02~1.05；会更容易识别为 curved。

CURVE_MAX_DEVIATION_RATIO = 0.025
# 曲线中间点偏离端点弦线的最大距离/总弧长阈值。用于识别轻微波浪，建议 0.015~0.05。

AXIS_TOLERANCE_DEG = 18.0
# 主方向距水平/竖直轴不超过此角度，分类为 horizontal 或 vertical；其余近直线分类为 diagonal。

LOW_LINEARITY_THRESHOLD = 0.78
# PCA 主轴线性度低于此值时，如果不是闭环/明显弧线，则分类为 mixed。

# ---- 曲线之间的关联分组参数 ----
BUILD_RELATION_GROUPS = True
# True：把相交或间距小于 GROUP_GAP_TOL 的不同曲线加入同一几何关系组。
# 注意：关系组不等于语义上的一个汉字；相邻汉字很近时可能被归为同组。

GROUP_GAP_TOL = 0.35
# 不同 Primitive 的几何关联距离阈值。越大越容易把相邻笔画/点/部件关联起来。
# 若输入尺寸很大，可先从 0.5~1.0 测试；若希望只分组真正接触的线条，可设为 0.02~0.1。

RELATION_GRID_CELL = 0.35
# 关系检测空间哈希网格尺寸。一般取 GROUP_GAP_TOL；过大可能增加候选对数量。

MAX_RELATION_PAIR_TESTS = 750000
# 关系段对精确距离检查次数上限。达到上限时输出 WARNING；极复杂曲线可能只得到部分关系。

NUMERICAL_EPSILON = 1e-9

start_time = time.perf_counter()
node = hou.pwd()
if not node.inputs():
    raise hou.NodeError("python_stroke_analysis_v16：请连接 python_curve_smooth 或 python_directional_stretch 的曲线输出。")
src = node.inputs()[0].geometry()
geo = node.geometry()
geo.clear()
geo.merge(src)


def safe_float(value, default=0.0):
    try:
        return float(value)
    except Exception:
        return float(default)


def xz_dist(a, b):
    dx = float(a[0]) - float(b[0])
    dz = float(a[2]) - float(b[2])
    return math.sqrt(dx * dx + dz * dz)


def polyline_length(pts, closed):
    if len(pts) < 2:
        return 0.0
    nseg = len(pts) if closed else len(pts) - 1
    return sum(xz_dist(pts[i], pts[(i + 1) % len(pts)]) for i in range(nseg))


def signed_area_xz(pts):
    if len(pts) < 3:
        return 0.0
    area2 = 0.0
    for i in range(len(pts)):
        a, b = pts[i], pts[(i + 1) % len(pts)]
        area2 += float(a[0]) * float(b[2]) - float(b[0]) * float(a[2])
    return 0.5 * area2


def point_segment_distance_sq_xz(p, a, b):
    dx, dz = float(b[0]) - float(a[0]), float(b[2]) - float(a[2])
    px, pz = float(p[0]) - float(a[0]), float(p[2]) - float(a[2])
    den = dx * dx + dz * dz
    if den <= 1e-16:
        return px * px + pz * pz
    t = max(0.0, min(1.0, (px * dx + pz * dz) / den))
    qx, qz = float(a[0]) + t * dx, float(a[2]) + t * dz
    ex, ez = float(p[0]) - qx, float(p[2]) - qz
    return ex * ex + ez * ez


def cross2(ax, az, bx, bz):
    return ax * bz - az * bx


def segments_distance_sq_xz(a, b, c, d):
    # 若相交，距离为 0；否则取 4 个端点到对方线段的最小距离。
    rx, rz = float(b[0]) - float(a[0]), float(b[2]) - float(a[2])
    sx, sz = float(d[0]) - float(c[0]), float(d[2]) - float(c[2])
    den = cross2(rx, rz, sx, sz)
    qx, qz = float(c[0]) - float(a[0]), float(c[2]) - float(a[2])
    if abs(den) > 1e-12:
        t = cross2(qx, qz, sx, sz) / den
        u = cross2(qx, qz, rx, rz) / den
        if -1e-10 <= t <= 1.0 + 1e-10 and -1e-10 <= u <= 1.0 + 1e-10:
            return 0.0
    return min(
        point_segment_distance_sq_xz(a, c, d),
        point_segment_distance_sq_xz(b, c, d),
        point_segment_distance_sq_xz(c, a, b),
        point_segment_distance_sq_xz(d, a, b),
    )


def classify_curve(pts, closed):
    n = len(pts)
    length = polyline_length(pts, closed)
    if n < 2 or length <= NUMERICAL_EPSILON:
        return {
            "kind": "degenerate", "orientation": 0.0, "linearity": 0.0,
            "ratio": 0.0, "max_dev_ratio": 0.0, "area": 0.0, "length": length,
        }

    # 主轴方向：XZ 平面协方差的主特征向量角度（解析二维 PCA）。
    xs = [float(p[0]) for p in pts]
    zs = [float(p[2]) for p in pts]
    mx, mz = sum(xs) / n, sum(zs) / n
    vx = sum((x - mx) ** 2 for x in xs) / n
    vz = sum((z - mz) ** 2 for z in zs) / n
    cv = sum((xs[i] - mx) * (zs[i] - mz) for i in range(n)) / n
    angle_rad = 0.5 * math.atan2(2.0 * cv, vx - vz)
    orientation = (math.degrees(angle_rad) + 180.0) % 180.0
    tr = vx + vz
    disc = math.sqrt(max(0.0, (vx - vz) ** 2 + 4.0 * cv * cv))
    lam1 = max(0.0, 0.5 * (tr + disc))
    lam2 = max(0.0, 0.5 * (tr - disc))
    linearity = (lam1 - lam2) / max(lam1 + lam2, 1e-12)

    area = abs(signed_area_xz(pts)) if closed else 0.0
    chord = xz_dist(pts[0], pts[-1]) if not closed else 0.0
    ratio = length / max(chord, 1e-10) if not closed else float("inf")
    # 闭合图形区分圆环/圆形与有明显折角的闭合多边形（例如“田”的外框）。
    closed_sharp_count = 0
    if closed and n >= 4:
        for i in range(n):
            prevp, curp, nextp = pts[(i - 1) % n], pts[i], pts[(i + 1) % n]
            ax, az = float(curp[0]) - float(prevp[0]), float(curp[2]) - float(prevp[2])
            bx, bz = float(nextp[0]) - float(curp[0]), float(nextp[2]) - float(curp[2])
            la, lb = math.hypot(ax, az), math.hypot(bx, bz)
            if la > 1e-10 and lb > 1e-10:
                turn = math.degrees(math.acos(max(-1.0, min(1.0, (ax * bx + az * bz) / (la * lb)))))
                if turn >= 35.0:
                    closed_sharp_count += 1
    max_dev = 0.0
    if not closed:
        ax, az = float(pts[0][0]), float(pts[0][2])
        bx, bz = float(pts[-1][0]), float(pts[-1][2])
        denom = math.sqrt((bx - ax) ** 2 + (bz - az) ** 2)
        if denom > 1e-10:
            for p in pts[1:-1]:
                # 点到无限弦线的距离，评价轻微波浪偏离。
                max_dev = max(max_dev, abs((bx - ax) * (az - float(p[2])) - (ax - float(p[0])) * (bz - az)) / denom)
    max_dev_ratio = max_dev / max(length, 1e-10)

    if closed:
        if area <= max(1e-6, length * length * 1e-4):
            kind = "closed_curve"
        elif closed_sharp_count >= 2:
            kind = "closed_polygon"
        else:
            kind = "ring"
    elif length <= SHORT_FEATURE_MAX_LENGTH:
        kind = "short_feature"
    elif ratio >= CURVE_PATH_CHORD_RATIO or max_dev_ratio >= CURVE_MAX_DEVIATION_RATIO:
        kind = "arc" if ratio >= 1.15 and max_dev_ratio < 0.10 else "curved"
    elif linearity < LOW_LINEARITY_THRESHOLD:
        kind = "mixed"
    else:
        ang = orientation % 180.0
        horizontal_diff = min(ang, 180.0 - ang)
        vertical_diff = abs(ang - 90.0)
        if horizontal_diff <= AXIS_TOLERANCE_DEG:
            kind = "horizontal"
        elif vertical_diff <= AXIS_TOLERANCE_DEG:
            kind = "vertical"
        elif 0.0 < ang < 90.0:
            kind = "diagonal_rising"
        else:
            kind = "diagonal_falling"
    return {
        "kind": kind, "orientation": orientation, "linearity": linearity,
        "ratio": ratio if math.isfinite(ratio) else -1.0,

def clamp(value, low, high):
    return max(low, min(high, value))


def vec_dist(a, b):

def cell3(pos, cell_size):
    return (int(math.floor(pos[0] / cell_size)),

def cell2_xz(pos, cell_size):
    return (int(math.floor(pos[0] / cell_size)), int(math.floor(pos[2] / cell_size)))


def cross2(ax, az, bx, bz):
    return ax * bz - az * bx


def station_distance(stroke, a, b):
    d = abs(float(a) - float(b))

def dedupe_sorted(values, tolerance=0.02):
    out = []
    for value in sorted(float(v) for v in values):
        if not out or abs(value - out[-1]) > tolerance:
            out.append(value)
    return out

def sample_original_arc(stroke, arc):
    L = stroke["length"]
    pts = stroke["pts"]
    if stroke["closed"]:
        arc = float(arc) % L
    else:
        arc = clamp(float(arc), 0.0, L)
    cum = stroke["cum"]
    seg_lengths = stroke["seg_lengths"]
    i = bisect.bisect_right(cum, arc) - 1
    i = max(0, min(i, len(seg_lengths) - 1))
    seg_len = seg_lengths[i]
    if seg_len <= 1e-12:
        return hou.Vector3(pts[i])
    t = clamp((arc - cum[i]) / seg_len, 0.0, 1.0)
    return pts[i] + (pts[(i + 1) % len(pts)] - pts[i]) * t


def sample_station(stroke, station):
    L = stroke["length"]
    pts = stroke["pts"]
    if stroke["closed"]:
        return sample_original_arc(stroke, float(station) % L)
    source_arc = float(station) - stroke["ext_start"]
    if source_arc < 0.0:
        tangent = pts[1] - pts[0]
        if tangent.length() <= 1e-10:
            return hou.Vector3(pts[0])
        return pts[0] - tangent.normalized() * (-source_arc)
    if source_arc > L:
        tangent = pts[-1] - pts[-2]
        if tangent.length() <= 1e-10:
            return hou.Vector3(pts[-1])
        return pts[-1] + tangent.normalized() * (source_arc - L)
    return sample_original_arc(stroke, source_arc)


def vertex_turn_angle(stroke, i):
    pts = stroke["pts"]
    n = len(pts)
    if n < 3:
        return 0.0
    w = max(1, int(CORNER_WINDOW))
    if stroke["closed"]:
        w = min(w, max(1, n // 4))
        before = pts[(i - w) % n]
        cur = pts[i]
        after = pts[(i + w) % n]
    else:
        if i <= 0 or i >= n - 1:
            return 0.0
        w = min(w, i, n - 1 - i)
        before = pts[i - w]
        cur = pts[i]
        after = pts[i + w]
    a = cur - before
    b = after - cur
    if a.length() < 1e-10 or b.length() < 1e-10:
        return 0.0
    dot = clamp(a.normalized().dot(b.normalized()), -1.0, 1.0)
    return math.degrees(math.acos(dot))

def merge_coverage(anchor, coverage, arcs_are_station=False):
    for sid, arc_values in coverage.items():
        target = anchor["coverage"].setdefault(int(sid), [])
        stroke = stroke_by_id[int(sid)]
        length = max(float(stroke["length"]), 1e-12)
        tol = 0.03
        for arc in arc_values:
            arc = float(arc)
            if stroke["closed"]:
                arc %= length
            is_duplicate = False
            for previous in target:
                delta = abs(arc - previous)
                if stroke["closed"]:
                    delta = min(delta, max(0.0, length - delta))
                if delta <= tol:
                    is_duplicate = True
                    break
            if not is_duplicate:
                target.append(arc)
        target.sort()


def add_anchor(pos, coverage, type_name, priority, angle=0.0, source="geometry"):
    pos = hou.Vector3(pos)
    key = cell3(pos, anchor_cell)
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            for dz in (-1, 0, 1):
                for aid in anchor_grid.get((key[0] + dx, key[1] + dy, key[2] + dz), []):
                    old = anchors[aid]
                    if vec_dist(old["pos"], pos) <= FEATURE_MERGE_TOL:
                        merge_coverage(old, coverage)
                        old["types"].add(type_name)
                        old_priority = int(old["priority"])
                        if int(priority) > old_priority:
                            old_key = cell3(old["pos"], anchor_cell)
                            new_pos = hou.Vector3(pos)
                            new_key = cell3(new_pos, anchor_cell)
                            old["pos"] = new_pos
                            if new_key != old_key:
                                old_bucket = anchor_grid.get(old_key, [])
                                if aid in old_bucket:
                                    old_bucket.remove(aid)
                                if not old_bucket and old_key in anchor_grid:
                                    del anchor_grid[old_key]
                                anchor_grid[new_key].append(aid)
                        old["priority"] = max(old_priority, int(priority))
                        old["priority_before_merge"] = max(int(old["priority_before_merge"]), int(priority))
                        old["angle"] = max(old["angle"], float(angle))
                        old["sources"].add(source)
                        return aid
    aid = len(anchors)
    anchors.append({
        "anchor_id": aid, "pos": pos, "coverage": {}, "types": set([type_name]),
        "priority": int(priority), "priority_before_merge": int(priority),
        "angle": float(angle), "sources": set([source]), "layer": None,
        "local_nudge": 0.0, "placed": False,
    })
    merge_coverage(anchors[-1], coverage)
    anchor_grid[key].append(aid)
    return aid


def segment_intersection(seg_a, seg_b):
    a, b, c, d = seg_a["a"], seg_a["b"], seg_b["a"], seg_b["b"]
    rx, rz = b[0] - a[0], b[2] - a[2]
    sx, sz = d[0] - c[0], d[2] - c[2]
    denom = cross2(rx, rz, sx, sz)
    if abs(denom) < 1e-10:
        return None
    qx, qz = c[0] - a[0], c[2] - a[2]
    t = cross2(qx, qz, sx, sz) / denom
    u = cross2(qx, qz, rx, rz) / denom
    eps = 1e-8
    if t < -eps or t > 1.0 + eps or u < -eps or u > 1.0 + eps:
        return None
    t = clamp(t, 0.0, 1.0)
    u = clamp(u, 0.0, 1.0)
    x = a[0] + rx * t
    z = a[2] + rz * t
    y_a = a[1] + (b[1] - a[1]) * t
    y_b = c[1] + (d[1] - c[1]) * u
    return hou.Vector3(x, (y_a + y_b) * 0.5, z), t, u


def point_segment_projection_xz(p, seg):
    a, b = seg["a"], seg["b"]
    dx, dz = b[0] - a[0], b[2] - a[2]
    denom = dx * dx + dz * dz
    if denom <= 1e-12:
        return 0.0, (p[0] - a[0]) ** 2 + (p[2] - a[2]) ** 2
    t = ((p[0] - a[0]) * dx + (p[2] - a[2]) * dz) / denom
    t = clamp(t, 0.0, 1.0)
    qx, qz = a[0] + dx * t, a[2] + dz * t
    return t, (p[0] - qx) ** 2 + (p[2] - qz) ** 2

                                reject = True
                                break
                        if reject:
                            break
                    if reject:
                        break

                continue

            accepted_bins[bin_id].append(arc)

        add_anchor(stroke["pts"][i], {stroke["sid"]: [arc]}, "corner", 70, angle, source="curve_corner")

def _outward_tangent(stroke, side):
    pts = stroke["pts"]
    if len(pts) < 2:
        return hou.Vector3((0.0, 0.0, 0.0))
    vec = (pts[0] - pts[1]) if side == 0 else (pts[-1] - pts[-2])
    return vec.normalized() if vec.length() > 1e-10 else hou.Vector3((0.0, 0.0, 0.0))


def _endpoint_pos(stroke, side):

def _ep_find(key):
    while _endpoint_parent[key] != key:
        _endpoint_parent[key] = _endpoint_parent[_endpoint_parent[key]]
        key = _endpoint_parent[key]
    return key

def _ep_union(a, b):
    ra, rb = _ep_find(a), _ep_find(b)
    if ra != rb:
        if ra > rb:
            ra, rb = rb, ra
        _endpoint_parent[rb] = ra

def _stroke_find(sid):
    sid = int(sid)
    while _stroke_parent[sid] != sid:
        _stroke_parent[sid] = _stroke_parent[_stroke_parent[sid]]
        sid = _stroke_parent[sid]
    return sid

    ra, rb = _stroke_find(a), _stroke_find(b)
    if ra != rb:
        if ra > rb:
            ra, rb = rb, ra
        _stroke_parent[rb] = ra

def infer_point_layer(pos):
    return 1 if abs(float(pos[1]) - AUX_PLANE_Y) < abs(float(pos[1]) - BASE_PLANE_Y) else 0


def find_conflicts(pos, candidate_layer=None):
    pos = hou.Vector3(pos)
    if candidate_layer is None:
        candidate_layer = infer_point_layer(pos)
    key = cell3(pos, point_cell_size)
    conflicts = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            for dz in (-1, 0, 1):
                for item in selected_grid.get((key[0] + dx, key[1] + dy, key[2] + dz), []):
                    delta = pos - item["pos"]
                    d2 = delta.dot(delta)
                    if d2 < REQUIRED_DISTANCE2 - NUMERICAL_EPSILON:
                        conflicts.append(item)
                        continue
                    if int(candidate_layer) == int(item["layer"]):
                        if d2 < SAME_PLANE_CENTER_CLEARANCE2 - NUMERICAL_EPSILON:
                            conflicts.append(item)
                    else:
                        dxz = float(pos[0] - item["pos"][0])
                        dzxz = float(pos[2] - item["pos"][2])
                        xz_distance = math.sqrt(dxz * dxz + dzxz * dzxz)
                        overlap = max(0.0, float(VISUAL_CIRCLE_DIAMETER) - xz_distance)
                        if overlap > MAX_CROSS_LAYER_VISUAL_OVERLAP_HARD + NUMERICAL_EPSILON:
                            conflicts.append(item)
    return conflicts


def _coverage_sids_of(item):
    return [int(s) for s in item.get("coverage", {}).keys()]


def _source_tier_allows_soft_placement(context_sids, item):
    if not ENABLE_CONFLICT_SOURCE_TIERING:
        return False
    for sid_a in context_sids:
        sa = stroke_by_id.get(int(sid_a))
        if sa is None:
            continue
        for sid_b in _coverage_sids_of(item):
            sb = stroke_by_id.get(int(sid_b))
            if sb is None:
                continue
            if int(sa.get("continuity_chain_id", sa["sid"])) == int(sb.get("continuity_chain_id", sb["sid"])):
                return True
            if abs(int(sa["prim"]) - int(sb["prim"])) <= int(STROKE_NEARBY_PRIM_RANGE):
                return True
            group_a = int(sa.get("shape_group_id", -1))
            group_b = int(sb.get("shape_group_id", -1))
            if group_a >= 0 and group_a == group_b:
                return True
    return False


def placement_allowed(pos, candidate_layer, context_sids):
    pos = hou.Vector3(pos)
    candidate_layer = int(candidate_layer)
    context_sids = set(int(s) for s in context_sids)
    key = cell3(pos, point_cell_size)
    soft_overlap_max = float(CONTROLLED_OVERLAP_MAX) * float(VISUAL_CIRCLE_DIAMETER)
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            for dz in (-1, 0, 1):
                for item in selected_grid.get((key[0] + dx, key[1] + dy, key[2] + dz), []):
                    delta = pos - item["pos"]
                    d2 = delta.dot(delta)
                    if d2 < REQUIRED_DISTANCE2 - NUMERICAL_EPSILON:
                        return False, item, "hard_3d"
                    if candidate_layer == int(item["layer"]):
                        if d2 < SAME_PLANE_CENTER_CLEARANCE2 - NUMERICAL_EPSILON:
                            if ENABLE_CONTROLLED_OVERLAP:
                                dxz = float(pos[0] - item["pos"][0])
                                dzxz = float(pos[2] - item["pos"][2])
                                xz_d = math.sqrt(dxz * dxz + dzxz * dzxz)
                                overlap = max(0.0, float(VISUAL_CIRCLE_DIAMETER) - xz_d)
                                if overlap <= soft_overlap_max + NUMERICAL_EPSILON and _source_tier_allows_soft_placement(context_sids, item):
                                    continue
                            return False, item, "same_plane_visual"
                    else:
                        dxz = float(pos[0] - item["pos"][0])
                        dzxz = float(pos[2] - item["pos"][2])
                        xz_distance = math.sqrt(dxz * dxz + dzxz * dzxz)
                        overlap = max(0.0, float(VISUAL_CIRCLE_DIAMETER) - xz_distance)
                        if overlap > MAX_CROSS_LAYER_VISUAL_OVERLAP_HARD + NUMERICAL_EPSILON:
                            return False, item, "cross_layer_hard"
    return True, None, ""


def fixed_auxiliary_position(base_pos):

def insert_point(pos, coverage, role, priority, anchor=None, layer=0, local_nudge=0.0, regular_target=None, stroke_id=None, depth_offset=0.0):
    global max_layer_used, auxiliary_anchor_count, regular_placed_count
    pos = hou.Vector3(pos)
    coverage_copy = {int(sid): [float(a) for a in arcs] for sid, arcs in coverage.items()}
    for cov_sid, cov_arcs in list(coverage_copy.items()):
        cov_stroke = stroke_by_id[cov_sid]
        if cov_stroke["closed"]:
            coverage_copy[cov_sid] = [float(a) % cov_stroke["length_eff"] for a in cov_arcs]
    if stroke_id is None:
        stroke_id = min(coverage_copy.keys()) if coverage_copy else -1
    if stroke_id < 0:
        raise hou.NodeError("内部错误：点没有关联到任何 stroke_id。")
    primary_sid = int(stroke_id)
    stroke = stroke_by_id[primary_sid]
    primary_stations = coverage_copy.get(primary_sid, [0.0])
    item = {
        "point_id": len(selected_items), "pos": pos, "coverage": coverage_copy,
        "role": role, "priority": int(priority), "layer": int(layer),
        "anchor_id": int(anchor["anchor_id"]) if anchor is not None else -1,
        "types": set(anchor["types"]) if anchor is not None else set(),
        "angle": float(anchor["angle"]) if anchor is not None else 0.0,
        "primary_sid": primary_sid, "primary_station": float(primary_stations[0]),
        "ideal_station": float(regular_target if regular_target is not None else primary_stations[0]),
        "local_nudge": float(local_nudge), "endpoint_extension": 0.0,
        "depth_offset": float(depth_offset),
    }
    if anchor is not None and "endpoint" in anchor["types"] and "junction" not in anchor["types"]:
        station = float(primary_stations[0])
        start_station = float(stroke["ext_start"])
        end_station = float(stroke["ext_start"] + stroke["length"])
        if abs(station - start_station) <= 0.05:
            item["endpoint_extension"] = float(stroke["ext_start"])
        elif abs(station - end_station) <= 0.05:
            item["endpoint_extension"] = float(stroke["ext_end"])
    selected_items.append(item)
    selected_grid[cell3(pos, point_cell_size)].append(item)
    selected_xz_grid[cell2_xz(pos, point_cell_size)].append(item)
    for covered_sid in coverage_copy.keys():
        items_by_stroke[int(covered_sid)].append(item)
    max_layer_used = max(max_layer_used, int(layer))
    if layer > 0:
        auxiliary_anchor_count += 1
    if role in ("regular", "gap_repair", "closed_regular", "junction_span_witness"):
        regular_placed_count += 1
    return item


def cross_layer_visual_overlap_stats(pos, candidate_layer, circle_diameter=None):
    diameter = float(VISUAL_CIRCLE_DIAMETER if circle_diameter is None else circle_diameter)
    key = cell2_xz(pos, point_cell_size)
    max_overlap = 0.0
    total_overlap = 0.0
    pair_count = 0
    min_xz = float("inf")
    for ox in (-1, 0, 1):
        for oz in (-1, 0, 1):
            for item in selected_xz_grid.get((key[0] + ox, key[1] + oz), []):
                if int(item["layer"]) == int(candidate_layer):
                    continue
                dx = float(pos[0] - item["pos"][0])
                dz = float(pos[2] - item["pos"][2])
                d = math.sqrt(dx * dx + dz * dz)
                min_xz = min(min_xz, d)
                overlap = max(0.0, diameter - d)
                if overlap > NUMERICAL_EPSILON:
                    pair_count += 1
                    max_overlap = max(max_overlap, overlap)
                    total_overlap += overlap
    if min_xz == float("inf"):
        min_xz = 1e9
    return max_overlap, total_overlap, pair_count, min_xz


def endpoint_nudge_options(anchor):
    original = (hou.Vector3(anchor["pos"]), 0.0)
    if LOCK_ENDPOINTS_EXACT or not ALLOW_LOCAL_ENDPOINT_NUDGE or "endpoint" not in anchor["types"] or "junction" in anchor["types"]:
        return [original]
    if len(anchor["coverage"]) != 1:
        return [original]
    sid = next(iter(anchor["coverage"].keys()))
    stroke = stroke_by_id[sid]
    stations = [float(v) for v in anchor["coverage"][sid]]
    station = min(stations, key=lambda v: min(abs(v), abs(stroke["length_eff"] - v)))
    start_station = float(stroke["ext_start"])
    end_station = float(stroke["ext_start"] + stroke["length"])
    if abs(station - start_station) <= 0.05:
        tangent = stroke["pts"][0] - stroke["pts"][1]
        already_used = stroke["ext_start"] + stroke["endpoint_local_used_start"]
    elif abs(station - end_station) <= 0.05:
        tangent = stroke["pts"][-1] - stroke["pts"][-2]
        already_used = stroke["ext_end"] + stroke["endpoint_local_used_end"]
    else:
        return [original]
    if tangent.length() <= 1e-10:
        return [original]
    tangent = tangent.normalized()
    free_single = max(0.0, MAX_SINGLE_ENDPOINT_EXTENSION - already_used)
    free_total = max(0.0, MAX_TOTAL_ENDPOINT_EXTENSION - stroke["ext_start"] - stroke["ext_end"] - stroke["endpoint_local_used_start"] - stroke["endpoint_local_used_end"])
    max_extra = min(MAX_LOCAL_ENDPOINT_NUDGE, free_single, free_total)
    nudges = []
    if max_extra > NUMERICAL_EPSILON and LOCAL_ENDPOINT_NUDGE_STEP > 0.0:
        steps = int(math.floor(max_extra / LOCAL_ENDPOINT_NUDGE_STEP + 1e-9))
        for k in range(1, steps + 1):
            delta = float(k) * LOCAL_ENDPOINT_NUDGE_STEP
            nudges.append((hou.Vector3(anchor["pos"] + tangent * delta), delta))
    return ([original] + nudges) if TRY_ORIGINAL_ENDPOINT_FIRST else (nudges + [original])


def record_endpoint_nudge(anchor, nudge):
    if nudge <= 0.0 or "endpoint" not in anchor["types"] or "junction" in anchor["types"] or len(anchor["coverage"]) != 1:
        return
    sid = next(iter(anchor["coverage"].keys()))
    stroke = stroke_by_id[sid]
    stations = [float(v) for v in anchor["coverage"][sid]]
    station = min(stations, key=lambda v: min(abs(v), abs(stroke["length_eff"] - v)))
    start_station = float(stroke["ext_start"])
    end_station = float(stroke["ext_start"] + stroke["length"])
    if abs(station - start_station) <= 0.05:
        stroke["endpoint_local_used_start"] += float(nudge)
    elif abs(station - end_station) <= 0.05:
        stroke["endpoint_local_used_end"] += float(nudge)

def _remove_item_from_3d_grid(item):
    key = cell3(item["pos"], point_cell_size)
    bucket = selected_grid.get(key, [])

    if not selected_grid[key]:
        selected_grid.pop(key, None)


def _move_selected_item_to_layer(item, target_layer):
    target_layer = int(target_layer)
    if target_layer == int(item["layer"]):
        return True
    old_pos = hou.Vector3(item["pos"])
    old_layer = int(item["layer"])
    old_offset = float(item.get("depth_offset", 0.0))
    _remove_item_from_3d_grid(item)
    target_y = BASE_PLANE_Y if target_layer == 0 else AUX_PLANE_Y
    new_pos = hou.Vector3(float(old_pos[0]), float(target_y), float(old_pos[2]))
    conflicts = find_conflicts(new_pos, candidate_layer=target_layer)
    if not conflicts:
        item["pos"] = new_pos
        item["layer"] = target_layer
        item["depth_offset"] = 0.0 if target_layer == 0 else float(AUX_LAYER_SIGN) * float(AUX_FIXED_OFFSET)
        selected_grid[cell3(new_pos, point_cell_size)].append(item)
        return True
    item["pos"] = old_pos
    item["layer"] = old_layer
    item["depth_offset"] = old_offset
    selected_grid[cell3(old_pos, point_cell_size)].append(item)
    return False


def try_place_anchor(anchor, allow_aux=False, allow_relocate_low_priority=False):
    global local_endpoint_nudge_count
    if anchor["placed"]:
        return True

    preferred_layer = int(anchor.get("preferred_layer", 0))
    if preferred_layer not in (0, 1):
        preferred_layer = 0
    layer_order = [preferred_layer]
    if allow_aux:
        other_layer = 1 - preferred_layer
        if other_layer == 0 or _anchor_is_aux_eligible(anchor):
            layer_order.append(other_layer)
    if preferred_layer == 1 and not ENABLE_FIXED_AUXILIARY_LAYER:
        layer_order = [0]

    options = endpoint_nudge_options(anchor)
    for layer in layer_order:
        if layer == 1 and not _anchor_is_aux_eligible(anchor):
            continue
        for candidate_base, nudge in options:
            if layer == 0:
                candidate_pos = hou.Vector3(candidate_base)
                depth_offset = 0.0
            else:
                candidate_pos, depth_offset = fixed_auxiliary_position(candidate_base)
                if candidate_pos is None:
                    continue

            if layer == 1:
                _max_overlap, _total_overlap, _overlap_pairs, _min_xz = cross_layer_visual_overlap_stats(candidate_pos, 1)
                if _max_overlap > MAX_STRUCTURAL_AUX_VISUAL_OVERLAP + NUMERICAL_EPSILON:
                    continue
            conflicts = find_conflicts(candidate_pos, candidate_layer=layer)
            if conflicts and allow_relocate_low_priority:
                can_displace = ("endpoint" in anchor.get("types", set()) or "junction" in anchor.get("types", set()))
                if can_displace:
                    for blocker in sorted(conflicts, key=lambda it: (it.get("priority", 0), it["point_id"])):
                        if blocker.get("role") != "anchor":
                            continue
                        blocker_types = blocker.get("types", set())
                        is_corner_only = ("corner" in blocker_types and "endpoint" not in blocker_types and "junction" not in blocker_types)
                        if is_corner_only and not ALLOW_RELOCATE_CORNER_TO_AUX_FOR_ANCHOR_RESCUE:
                            continue
                        if not is_corner_only and int(blocker.get("priority", 0)) >= int(anchor.get("priority", 0)):
                            continue
                        _move_selected_item_to_layer(blocker, 1 - int(blocker["layer"]))
                    conflicts = find_conflicts(candidate_pos, candidate_layer=layer)
            if conflicts:
                continue

            cov = anchor["coverage"]
            item = insert_point(candidate_pos, cov, "anchor", anchor["priority"], anchor,
                                layer, nudge, depth_offset=depth_offset)
            anchor["placed"] = True
            anchor["layer"] = layer
            anchor["local_nudge"] = nudge
            anchor["failure"] = ""
            if nudge > 0.0:
                local_endpoint_nudge_count += 1
                record_endpoint_nudge(anchor, nudge)
            return True

    anchor["placed"] = False
    if preferred_layer == 1:
        anchor["failure"] = "PREFERRED_AUX_LAYER_BLOCKED"
    else:
        anchor["failure"] = "NO_FEASIBLE_LAYER_POSITION"
    return False


def assign_minimum_aux_layers_to_structural_anchors(anchor_list):
    ids = [i for i, a in enumerate(anchor_list)
           if ("endpoint" in a.get("types", set()) or "junction" in a.get("types", set()))]
    if not ids:
        return {"components": 0, "bipartite_components": 0, "non_bipartite_components": 0, "preassigned_aux": 0}

    cell = max(float(SAME_PLANE_CENTER_CLEARANCE), 1e-6)
    buckets = defaultdict(list)
    adjacency = {i: set() for i in ids}
    for i in ids:
        pos = anchor_list[i]["pos"]
        key = cell2_xz(pos, cell)
        for dx in (-1, 0, 1):
            for dz in (-1, 0, 1):
                for j in buckets.get((key[0] + dx, key[1] + dz), []):
                    other = anchor_list[j]["pos"]
                    ddx = float(pos[0] - other[0])
                    ddz = float(pos[2] - other[2])
                    if ddx * ddx + ddz * ddz < SAME_PLANE_CENTER_CLEARANCE2 - NUMERICAL_EPSILON:
                        adjacency[i].add(j)
                        adjacency[j].add(i)
        buckets[key].append(i)

    def auxiliary_cost(anchor):
        types = anchor.get("types", set())
        cover_count = max(1, len(anchor.get("coverage", {})))
        if "junction" in types and "endpoint" in types:
            return 24.0 / cover_count
        if "junction" in types:
            return 36.0 / cover_count
        if "endpoint" in types:
            return 100.0
        return 50.0

    colors = {}
    component_count = 0
    bipartite_count = 0
    non_bipartite_count = 0
    preassigned_aux = 0
    for root in ids:
        if root in colors:
            continue
        component_count += 1
        colors[root] = 0
        stack = [root]
        comp = []
        bipartite = True
        while stack:
            cur = stack.pop()
            comp.append(cur)
            for nxt in adjacency[cur]:
                if nxt not in colors:
                    colors[nxt] = 1 - colors[cur]
                    stack.append(nxt)
                elif colors[nxt] == colors[cur]:
                    bipartite = False
        if not bipartite:
            non_bipartite_count += 1
            for i in comp:
                anchor_list[i]["preferred_layer"] = 0
                anchor_list[i]["anchor_conflict_component_nonbipartite"] = True
            continue
        bipartite_count += 1
        part0 = [i for i in comp if colors[i] == 0]
        part1 = [i for i in comp if colors[i] == 1]
        def partition_score(part):
            return (len(part), sum(auxiliary_cost(anchor_list[i]) for i in part), tuple(sorted(part)))
        aux_color = 0 if partition_score(part0) < partition_score(part1) else 1
        for i in comp:
            layer = 1 if colors[i] == aux_color else 0
            anchor_list[i]["preferred_layer"] = layer
            if layer == 1:
                preassigned_aux += 1
    for i, a in enumerate(anchor_list):
        a.setdefault("preferred_layer", 0)
    return {"components": component_count, "bipartite_components": bipartite_count,
            "non_bipartite_components": non_bipartite_count, "preassigned_aux": preassigned_aux}

def stations_for_stroke(stroke, roles=None):
    sid = stroke["sid"]
    values = []
    for item in items_by_stroke.get(sid, []):
        if roles is not None and item["role"] not in roles:
            continue
        values.extend(item["coverage"].get(sid, []))
    if stroke["closed"]:
        L = stroke["length_eff"]
        values = [v % L for v in values]
    else:
        L = stroke["length_eff"]
        values = [clamp(v, 0.0, L) for v in values]
    return dedupe_sorted(values, 0.02)


def _estimate_regular_slots_for_spacing(spacing):
    spacing = max(0.01, float(spacing))
    total = 0
    for _stroke in strokes:
        _sid = _stroke["sid"]
        _L = float(_stroke["length_eff"])
        if _L <= 1e-8:
            continue
        _anchors_here = stations_for_stroke(_stroke, roles={"anchor"})
        if _stroke["closed"] and not _anchors_here:
            _target_segments = max(1, int(math.floor(_L / spacing + 0.5)))
            _safe_segments = max(1, int(math.floor(_L / SAME_PLANE_CENTER_CLEARANCE + 1e-8)))
            total += max(1, min(_target_segments, _safe_segments))
            continue
        if _stroke["closed"]:
            _bounds = sorted(_anchors_here)
            _intervals = []
            for _i, _a in enumerate(_bounds):
                _b = _bounds[(_i + 1) % len(_bounds)]
                if _i == len(_bounds) - 1:
                    _b += _L
                _intervals.append((_a, _b))
        else:
            _bounds = dedupe_sorted([0.0] + list(_anchors_here) + [_L], 0.02)
            _intervals = [(_bounds[_i], _bounds[_i + 1]) for _i in range(len(_bounds) - 1)]
        for _a, _b in _intervals:
            _gap = float(_b - _a)
            if _stroke["closed"] and _gap <= 0.0:
                _gap += _L
            _target_segments = max(1, int(math.floor(_gap / spacing + 0.5)))
            _safe_segments = max(1, int(math.floor(_gap / SAME_PLANE_CENTER_CLEARANCE + 1e-8)))
            _nseg = max(1, min(_target_segments, _safe_segments))
            total += max(0, _nseg - 1)
    return int(total)


def _count_potential_internal_witnesses():
    _count = 0
    for _stroke in strokes:
        if not ENSURE_JUNCTION_SPAN_WITNESS or not _stroke.get("is_internal_cross_stroke", False):
            continue
        _L = float(_stroke["length_eff"])
        if _L < MIN_INTERNAL_CROSS_STROKE_LENGTH:
            continue
        _vals = stations_for_stroke(_stroke)
        _margin = min(0.12, _L * 0.05)
        _has_interior = any((v > _margin and v < _L - _margin) for v in _vals)
        if not _has_interior:
            _count += 1
    return _count

def split_gap_count(gap):
    if gap <= 1e-8:
        return 1
    target_segments = max(1, int(math.floor(gap / TARGET_SPACING + 0.5)))
    safe_segments = int(math.floor(gap / SAME_PLANE_CENTER_CLEARANCE + 1e-8))
    if safe_segments < 1:
        return 1
    return max(1, min(target_segments, safe_segments))


def make_candidate_station(stroke, a, b, fraction):
    gap = b - a
    station = a + gap * fraction
    if stroke["closed"]:
        station %= stroke["length_eff"]
    return station


def gap_candidates_for_stroke(stroke):
    sid = stroke["sid"]
    L = stroke["length_eff"]
    anchors_here = stations_for_stroke(stroke, roles={"anchor"})
    if stroke["closed"] and not anchors_here:
        nseg = split_gap_count(L)
        if nseg <= 0:
            return []
        # 修复：interval 必须是 (a, b) 二元组；扩到 (-L, 2L) 使 search 在 0 附近可双向 wrap。
        return [(L * i / float(nseg), L, (-L, 2.0 * L), "closed_regular") for i in range(nseg)]

    if stroke["closed"]:
        if not anchors_here:
            bounds = [0.0, L]
        else:
            bounds = sorted(anchors_here)
            intervals = []
            for i in range(len(bounds)):
                a = bounds[i]
                b = bounds[(i + 1) % len(bounds)]
                if i == len(bounds) - 1:
                    b += L
                intervals.append((a, b))
    else:
        bounds = dedupe_sorted([0.0] + anchors_here + [L], 0.02)
        intervals = [(bounds[i], bounds[i + 1]) for i in range(len(bounds) - 1)]

    if stroke["closed"] and anchors_here:
        pass
    elif not stroke["closed"]:
        intervals = [(bounds[i], bounds[i + 1]) for i in range(len(bounds) - 1)]

    result = []
    for a, b in intervals:
        gap = b - a
        if stroke["closed"] and gap <= 0.0:
            gap += L
        nseg = split_gap_count(gap)
        if nseg <= 1:
            continue
        for k in range(1, nseg):
            station = a + gap * k / float(nseg)
            if not stroke["closed"]:
                station = clamp(station, 0.0, L)
            result.append((station, gap, (a, b), "regular"))
    return result

def reuse_coincident_point(pos, sid, station):
    key = cell2_xz(pos, point_cell_size)
    tol2 = COINCIDENT_POINT_TOL * COINCIDENT_POINT_TOL
    for ox in (-1, 0, 1):
        for oz in (-1, 0, 1):
            for item in selected_xz_grid.get((key[0] + ox, key[1] + oz), []):
                dx = float(pos[0] - item["pos"][0])
                dz = float(pos[2] - item["pos"][2])
                dy = float(pos[1] - item["pos"][1])
                if dx * dx + dz * dz <= tol2 and abs(dy) <= COINCIDENT_POINT_TOL:
                    stroke = stroke_by_id[int(sid)]
                    value = float(station)
                    if stroke["closed"]:
                        value %= stroke["length_eff"]
                    else:
                        value = clamp(value, 0.0, stroke["length_eff"])
                    values = item["coverage"].setdefault(int(sid), [])
                    if not any(station_distance(stroke, value, old) <= 0.03 for old in values):
                        values.append(value)
                        values.sort()
                    if not any(old_item is item for old_item in items_by_stroke[int(sid)]):
                        items_by_stroke[int(sid)].append(item)
                    return item
    return None


def regular_station_options(candidate):
    stroke = candidate["stroke"]
    target = float(candidate["station"])
    closed = stroke["closed"]
    L = stroke["length_eff"]
    a, b = candidate["interval"]
    lo = min(float(a), float(b) if float(b) >= float(a) else float(b) + (L if closed else 0.0)) + 0.03
    hi = max(float(a), float(b) if float(b) >= float(a) else float(b) + (L if closed else 0.0)) - 0.03
    if hi < lo:
        return [target % L] if closed else [clamp(target, 0.0, L)]

    raw = [target]
    for k in range(1, max(0, int(SEARCH_ARC_STEPS)) + 1):
        delta = float(SEARCH_ARC_STEP) * k
        raw.extend([target + delta, target - delta])

    if ENABLE_ADAPTIVE_LOCAL_SEARCH:
        gap_width = max(0.0, hi - lo)
        max_shift = gap_width * 0.45
        for fraction in ADAPTIVE_SEARCH_FRACTIONS:
            delta = min(gap_width * float(fraction), max_shift)
            if delta > 1e-8:
                raw.extend([target + delta, target - delta])

    valid = []
    seen = set()
    for value in raw:
        if value < lo - 1e-8 or value > hi + 1e-8:
            continue
        if closed:
            value = value % L
        else:
            value = clamp(value, 0.0, L)
        key = round(float(value), 6)
        if key in seen:
            continue
        seen.add(key)
        valid.append(float(value))

    valid.sort(key=lambda value: (station_distance(stroke, value, target), value))
    return valid[:max(1, int(MAX_LOCAL_SEARCH_OPTIONS))]

def auxiliary_budget_blocks_regular_gap(candidate):
    if not AUXILIARY_SOFT_BUDGET_ACTIVE:
        return False
    if candidate.get("role") != "gap_repair":
        return False
    aux_now = sum(1 for it in selected_items if int(it.get("layer", 0)) > 0)
    soft_target = auxiliary_soft_target_count(len(selected_items) + 1)
    if aux_now < soft_target:
        return False
    gap = float(candidate.get("gap", 0.0))
    if ENABLE_LARGE_GAP_AUX_PRIORITY and gap >= float(TARGET_SPACING) * max(1.0, float(VISUAL_GAP_TRIGGER_RATIO)):
        return False
    emergency = float(TARGET_SPACING) * max(1.0, float(AUXILIARY_OVER_BUDGET_GAP_RATIO))
    return gap < emergency


def place_regular_candidate(candidate, allow_aux=False):
    global regular_failed_count, regular_shifted_count, regular_aux_count, regular_reused_count
    global regular_visual_overlap_blocked_count, regular_controlled_overlap_count, aux_gap_repair_budget_blocked_count, point_budget_blocked_count, auxiliary_soft_budget_blocked_count
    stroke = candidate["stroke"]
    sid = candidate["sid"]
    opts = regular_station_options(candidate)

    if TARGET_COUNT_MODE_ACTIVE and len(selected_items) >= int(TARGET_POINT_COUNT):
        for station in opts:
            pos = sample_station(stroke, station)
            reused = reuse_coincident_point(pos, sid, station)
            if reused is not None:
                regular_reused_count += 1
                return True
        point_budget_blocked_count += 1
        regular_failed_count += 1
        return False

    for station in opts:
        pos = sample_station(stroke, station)
        reused = reuse_coincident_point(pos, sid, station)
        if reused is not None:
            regular_reused_count += 1
            return True
        allowed, _blocker, _kind = placement_allowed(pos, 0, [sid])
        if allowed:
            if find_conflicts(pos, candidate_layer=0):
                regular_controlled_overlap_count += 1
            insert_point(pos, {sid: [station]}, candidate["role"], 20, None, 0, 0.0, candidate["station"], sid)
            if station_distance(stroke, station, candidate["station"]) > 1e-5:
                regular_shifted_count += 1
            return True

    aux_allowed_for_role = False
    if candidate.get("role") == "junction_span_witness" and ALLOW_AUX_FOR_JUNCTION_SPAN_WITNESS:
        aux_allowed_for_role = True
    elif candidate.get("role") == "gap_repair" and ALLOW_AUXILIARY_GAP_REPAIR:
        large_enough = float(candidate.get("gap", 0.0)) >= float(TARGET_SPACING) * max(1.0, AUX_GAP_TRIGGER_RATIO)
        aux_allowed_for_role = large_enough
    elif candidate.get("role") == "closed_regular" and ENABLE_AUX_FOR_CLOSED_QUOTA:
        _closed_stroke = candidate["stroke"]
        _closed_current = len(items_by_stroke.get(int(candidate["sid"]), []))
        _closed_quota = max(int(STROKE_QUOTA_MIN_POINTS), int(math.ceil(float(_closed_stroke["length_eff"]) / float(CLOSED_QUOTA_DIVISOR))))
        aux_allowed_for_role = bool(ENABLE_CLOSED_CURVE_QUOTA and _closed_current < _closed_quota)

    if allow_aux and ENABLE_REGULAR_POINT_AUXILIARY and aux_allowed_for_role:
        if auxiliary_budget_blocks_regular_gap(candidate):
            auxiliary_soft_budget_blocked_count += 1
            regular_failed_count += 1
            return False
        if candidate.get("role") == "gap_repair":
            chain_id = int(stroke.get("continuity_chain_id", sid))
            chain_length = float(stroke.get("continuity_chain_length", stroke["length_eff"]))
            dynamic_budget = max(int(MAX_AUX_GAP_REPAIRS_PER_CHAIN_BASE),
                                 int(math.ceil(float(AUX_GAP_CHAIN_BUDGET_SCALE) * chain_length / max(TARGET_SPACING * AUX_GAP_CHAIN_LENGTH_DIVISOR, REQUIRED_DISTANCE * 1.5))))
            if aux_gap_repair_count_by_chain[chain_id] >= dynamic_budget:
                aux_gap_repair_budget_blocked_count += 1
                regular_failed_count += 1
                return False

        gap = max(0.0, float(candidate.get("gap", 0.0)))
        ideal_overlap_floor = max(0.0, float(VISUAL_CIRCLE_DIAMETER) - 0.5 * gap)
        preferred_overlap = float(PREFERRED_REGULAR_AUX_VISUAL_OVERLAP)
        relaxed_overlap = min(float(MAX_REGULAR_AUX_VISUAL_OVERLAP), ideal_overlap_floor + 0.05)
        if gap < float(TARGET_SPACING) * float(AUX_OVERLAP_RELAX_GAP_RATIO):
            relaxed_overlap = min(relaxed_overlap, preferred_overlap)
        # V16 staged 机制 j: 大缺口允许重叠上限修正（gap 大时不再被 ideal_overlap_floor 压到近零）
        if ENABLE_STAGED_RELAXED_OVERLAP_CAP and candidate.get("role") == "gap_repair":
            if gap >= float(TARGET_SPACING) * max(1.0, float(STAGED_AUX_OVERLAP_MIN_GAP_RATIO)):
                relaxed_overlap = min(float(MAX_REGULAR_AUX_VISUAL_OVERLAP),
                                      max(relaxed_overlap, float(STAGED_AUX_OVERLAP_CAP)))

        max_overlap_allowed = (MAX_STRUCTURAL_AUX_VISUAL_OVERLAP
                              if candidate.get("role") == "junction_span_witness"
                              else relaxed_overlap)
        # V16 staged 机制 h: gap_repair 多分位站点点位（只在辅助层搜索阶段扩展，主平面循环仍用 opts）
        aux_stations = list(opts)
        if ENABLE_STAGED_GAP_MULTI_FRACTION and candidate.get("role") == "gap_repair":
            _a0, _b0 = candidate["interval"]
            _gap_w = max(0.0, float(_b0) - float(_a0))
            if stroke["closed"] and _gap_w > float(stroke["length_eff"]):
                _gap_w = float(stroke["length_eff"])
            for _frac in STAGED_GAP_MULTI_FRACTIONS:
                _st = float(_a0) + _gap_w * float(_frac)
                if stroke["closed"]:
                    _st %= stroke["length_eff"]
                else:
                    _st = clamp(_st, 0.0, stroke["length_eff"])
                _key = round(float(_st), 6)
                if not any(round(float(v), 6) == _key for v in aux_stations):
                    aux_stations.append(float(_st))
            aux_stations.sort(key=lambda v: (station_distance(stroke, v, float(candidate["station"])), v))
            aux_stations = aux_stations[:max(1, int(MAX_LOCAL_SEARCH_OPTIONS))]

        # V16 staged 机制 i: 辅助层位置生成（正上方投影优先；被硬冲突拦截时横向小偏移）
        def _staged_aux_position(base_pos):
            _pos, _dep = fixed_auxiliary_position(base_pos)
            if _pos is not None and not find_conflicts(_pos, candidate_layer=1):
                return _pos, float(_dep)
            if not ENABLE_AUX_LATERAL_OFFSET:
                return None, 0.0
            for _ang in range(0, 360, max(15, int(AUX_LATERAL_OFFSET_STEP_ANGLE))):
                _rad = math.radians(float(_ang))
                _offpos = (float(base_pos[0]) + float(AUX_LATERAL_OFFSET_DIST) * math.cos(_rad),
                           float(AUX_PLANE_Y),
                           float(base_pos[2]) + float(AUX_LATERAL_OFFSET_DIST) * math.sin(_rad))
                if find_conflicts(_offpos, candidate_layer=1):
                    continue
                return _offpos, float(AUX_LAYER_SIGN) * float(AUX_FIXED_OFFSET)
            return None, 0.0

        rescue = []
        low_overlap_rescue = []
        for station in aux_stations:
            base_pos = sample_station(stroke, station)
            pos, depth_offset = _staged_aux_position(base_pos)
            if pos is None:
                continue
            if find_conflicts(pos, candidate_layer=1):
                continue
            max_overlap, total_overlap, overlap_count, min_xz = cross_layer_visual_overlap_stats(pos, 1)
            row = (max_overlap, total_overlap, overlap_count,
                   abs(station - candidate["station"]), station, pos, float(depth_offset))
            if max_overlap <= preferred_overlap + NUMERICAL_EPSILON:
                low_overlap_rescue.append(row)
            if max_overlap <= max_overlap_allowed + NUMERICAL_EPSILON:
                rescue.append(row)
            else:
                regular_visual_overlap_blocked_count += 1

        chosen_pool = low_overlap_rescue if low_overlap_rescue else rescue
        if chosen_pool:
            _max_overlap, _total_overlap, _overlap_count, _station_delta, station, pos, depth_offset = min(
                chosen_pool, key=lambda row: (row[0], row[1], row[2], row[3]))
            layer = depth_band_from_offset(depth_offset)
            insert_point(pos, {sid: [station]}, candidate["role"], 15, None, layer, 0.0, candidate["station"], sid, depth_offset=depth_offset)
            regular_aux_count += 1
            if candidate.get("role") == "gap_repair":
                aux_gap_repair_count_by_chain[int(stroke.get("continuity_chain_id", sid))] += 1
            return True

    regular_failed_count += 1
    return False

def _stroke_quota(stroke):
    L = float(stroke["length_eff"])
    if stroke["closed"] and ENABLE_CLOSED_CURVE_QUOTA:
        return max(int(STROKE_QUOTA_MIN_POINTS), int(math.ceil(L / float(CLOSED_QUOTA_DIVISOR))))
    return max(int(STROKE_QUOTA_MIN_POINTS), int(math.ceil(L / (float(TARGET_SPACING) * float(STROKE_QUOTA_DIVISOR)))))


def place_regular_round_robin():
    if not ENABLE_STROKE_QUOTA_ROUND_ROBIN:
        for candidate in all_regular_candidates:
            place_regular_candidate(candidate, allow_aux=False)
        return
    candidates_by_sid = defaultdict(list)
    for candidate in all_regular_candidates:
        candidates_by_sid[int(candidate["sid"])].append(candidate)
    pending_sids = sorted(candidates_by_sid.keys())
    changed = True
    while changed and pending_sids:
        changed = False
        for sid in pending_sids[:]:
            stroke = stroke_by_id[int(sid)]
            quota = _stroke_quota(stroke)
            current = len(items_by_stroke.get(int(sid), []))
            if current >= quota:
                pending_sids.remove(sid)
                continue
            before = len(selected_items)
            placed_any = False
            for candidate in candidates_by_sid[int(sid)]:
                if place_regular_candidate(candidate, allow_aux=False):
                    placed_any = True
                    break
            if placed_any:
                changed = True
                if len(selected_items) == before:
                    pending_sids.remove(sid)
            else:
                pending_sids.remove(sid)

def stroke_gap_values(stroke, arcs):
    L = stroke["length_eff"]
    arcs = dedupe_sorted([a % L if stroke["closed"] else clamp(a, 0.0, L) for a in arcs], 0.02)
    if not arcs:
        return [L]
    if stroke["closed"]:
        if len(arcs) == 1:
            return [L]
        gaps = [arcs[i + 1] - arcs[i] for i in range(len(arcs) - 1)]
        gaps.append(L - arcs[-1] + arcs[0])
        return gaps
    bounds = dedupe_sorted([0.0] + arcs + [L], 0.02)
    return [bounds[i + 1] - bounds[i] for i in range(len(bounds) - 1)]


def diagnose_gap_blockers(stroke, arcs):
    sid = stroke["sid"]
    L = stroke["length_eff"]
    arcs = dedupe_sorted([a % L if stroke["closed"] else clamp(a, 0.0, L) for a in arcs], 0.02)
    if stroke["closed"]:
        if not arcs:
            intervals = [(0.0, L)]
        else:
            intervals = []
            for i, a in enumerate(arcs):
                b = arcs[(i + 1) % len(arcs)]
                if i == len(arcs) - 1:
                    b += L
                intervals.append((a, b))
    else:
        bounds = dedupe_sorted([0.0] + arcs + [L], 0.02)
        intervals = [(bounds[i], bounds[i + 1]) for i in range(len(bounds) - 1)]
    same_hits = other_hits = free_hits = 0
    blocker_ids = set()
    for a, b in intervals:
        gap = b - a
        if gap <= VISUAL_GAP_THRESHOLD:
            continue
        for frac in (0.25, 0.5, 0.75):
            station = a + gap * frac
            pos = sample_station(stroke, station)
            conflicts = find_conflicts(pos, candidate_layer=0)
            if not conflicts:
                free_hits += 1
                continue
            for item in conflicts:
                blocker_ids.add(item["point_id"])
                if sid in item["coverage"]:
                    same_hits += 1
                else:
                    other_hits += 1
    if same_hits and other_hits:
        cause = "MIXED_SAME_AND_OTHER_STROKES"
    elif same_hits:
        cause = "SAME_STROKE_POINTS"
    elif other_hits:
        cause = "OTHER_STROKES"
    elif free_hits:
        cause = "NO_BLOCKER_AT_SAMPLES__CHECK_TARGET_COUNT_OR_ANCHOR_SPACING"
    else:
        cause = "NOT_TESTED"
    blocker_strokes = set()
    for item in selected_items:
        if item["point_id"] in blocker_ids:
            blocker_strokes.update(item["coverage"].keys())
    return cause, sorted(blocker_ids), sorted(blocker_strokes), same_hits, other_hits, free_hits

import hou
import math
import time
from collections import defaultdict

# ============================================================================
# PARAMETERS — V16 final validator
# The upstream V16 spacing node writes effective values as detail attributes.
# ============================================================================
FALLBACK_MIN_DISTANCE = 2.0
FALLBACK_SAFETY_BUFFER = 0.0
FALLBACK_CIRCLE_DIAMETER = 2.0
FALLBACK_SAME_PLANE_EXTRA_GAP = 0.0
FALLBACK_AUX_LAYER_SIGN = +1
# Must match the project viewport convention: +Y is treated as "back" for V16.
MAX_CROSS_LAYER_VISUAL_OVERLAP_HARD = 0.35
NUMERICAL_EPSILON = 1e-7
PRINT_CONFLICT_PAIRS = True
MAX_PRINTED_PAIRS = 30

start = time.perf_counter()
node = hou.pwd()
if not node.inputs():
    raise hou.NodeError("python_validate_points_final_v16：请连接 python_global_spacing_v16 的输出。")
src = node.inputs()[0].geometry()
geo = node.geometry()
geo.clear()
geo.merge(src)
points = list(geo.points())


def detail(name, default=None):
    attrib = geo.findGlobalAttrib(name)
    if attrib is None:
        return default
    try:
        return geo.attribValue(attrib)
    except Exception:
        return default

required = float(detail("required_point_distance", FALLBACK_MIN_DISTANCE + max(0.0, FALLBACK_SAFETY_BUFFER)))
circle_diameter = float(detail("visual_circle_diameter", FALLBACK_CIRCLE_DIAMETER))
same_plane_clearance = float(detail("same_plane_center_clearance", max(required, circle_diameter + FALLBACK_SAME_PLANE_EXTRA_GAP)))
aux_offset = float(detail("auxiliary_fixed_offset", required + 0.05))
base_y = float(detail("base_plane_y", 0.0))
aux_y = float(detail("auxiliary_plane_y", base_y + FALLBACK_AUX_LAYER_SIGN * aux_offset))
cell_size = max(required, same_plane_clearance, circle_diameter, aux_offset, 1e-6)

# Ensure diagnostic point attributes exist; do not modify point positions.
safety_attr = geo.findPointAttrib("safety_conflict_count")
if safety_attr is None:
    safety_attr = geo.addAttrib(hou.attribType.Point, "safety_conflict_count", 0)
same_plane_attr = geo.findPointAttrib("final_same_plane_spacing_conflict_count")
if same_plane_attr is None:
    same_plane_attr = geo.addAttrib(hou.attribType.Point, "final_same_plane_spacing_conflict_count", 0)
projection_attr = geo.findPointAttrib("final_cross_layer_projection_overlap_count")
if projection_attr is None:
    projection_attr = geo.addAttrib(hou.attribType.Point, "final_cross_layer_projection_overlap_count", 0)
for p in points:
    p.setAttribValue(safety_attr, 0)
    p.setAttribValue(same_plane_attr, 0)
    p.setAttribValue(projection_attr, 0)
positions = [hou.Vector3(p.position()) for p in points]


def layer_of(p, pos):
    attrib = geo.findPointAttrib("depth_layer")
    if attrib is not None:
        try:
            return int(p.attribValue(attrib))
        except Exception:
            pass
    return 1 if abs(float(pos[1]) - aux_y) < abs(float(pos[1]) - base_y) else 0

layers = [layer_of(p, pos) for p, pos in zip(points, positions)]

def key3(pos):
    return (int(math.floor(pos[0] / cell_size)),
            int(math.floor(pos[1] / cell_size)),
            int(math.floor(pos[2] / cell_size)))

grid = defaultdict(list)
physical_violations = []
same_plane_violations = []
cross_layer_overlaps = []
cross_layer_hard_violations = []
min_distance = float("inf")
min_cross_layer_xz = float("inf")
max_projection_overlap = 0.0
pairs_checked = 0
for i, pos in enumerate(positions):
    key = key3(pos)
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            for dz in (-1, 0, 1):
                for j in grid.get((key[0] + dx, key[1] + dy, key[2] + dz), []):
                    other = positions[j]
                    pairs_checked += 1
                    d = (pos - other).length()
                    min_distance = min(min_distance, d)
                    if d < required - NUMERICAL_EPSILON:
                        physical_violations.append((j, i, d))
                        points[j].setAttribValue(safety_attr, int(points[j].attribValue(safety_attr)) + 1)
                        points[i].setAttribValue(safety_attr, int(points[i].attribValue(safety_attr)) + 1)
                    dxz = float(pos[0] - other[0])
                    dzxz = float(pos[2] - other[2])
                    xz = math.sqrt(dxz * dxz + dzxz * dzxz)
                    if layers[i] == layers[j]:
                        if xz < same_plane_clearance - NUMERICAL_EPSILON:
                            same_plane_violations.append((j, i, xz))
                            points[j].setAttribValue(same_plane_attr, int(points[j].attribValue("final_same_plane_spacing_conflict_count")) + 1)
                            points[i].setAttribValue(same_plane_attr, int(points[i].attribValue("final_same_plane_spacing_conflict_count")) + 1)
    grid[key].append(i)

if not math.isfinite(min_distance):
    min_distance = 0.0

# 独立 XZ 投影索引：辅助层和主层的 Y 不同，物理三维距离无法代表正面可读性。
# 因此跨层投影重叠按硬上限单独检查，超过阈值会计入失败。
projection_cell = max(circle_diameter, 1e-6)
projection_grid = defaultdict(list)
for i, pos in enumerate(positions):
    key2 = (int(math.floor(pos[0] / projection_cell)), int(math.floor(pos[2] / projection_cell)))
    for dx in (-1, 0, 1):
        for dz in (-1, 0, 1):
            for j in projection_grid.get((key2[0] + dx, key2[1] + dz), []):
                if layers[i] == layers[j]:
                    continue
                dxz = float(pos[0] - positions[j][0])
                dzxz = float(pos[2] - positions[j][2])
                xz = math.sqrt(dxz * dxz + dzxz * dzxz)
                min_cross_layer_xz = min(min_cross_layer_xz, xz)
                overlap = max(0.0, circle_diameter - xz)
                if overlap > NUMERICAL_EPSILON:
                    cross_layer_overlaps.append((j, i, xz, overlap))
                    if overlap > MAX_CROSS_LAYER_VISUAL_OVERLAP_HARD + NUMERICAL_EPSILON:
                        cross_layer_hard_violations.append((j, i, xz, overlap))
                    max_projection_overlap = max(max_projection_overlap, overlap)
                    points[j].setAttribValue(projection_attr, int(points[j].attribValue("final_cross_layer_projection_overlap_count")) + 1)
                    points[i].setAttribValue(projection_attr, int(points[i].attribValue("final_cross_layer_projection_overlap_count")) + 1)
    projection_grid[key2].append(i)

def json_safe(value, depth=0):
    """Convert HOM values and nested attributes into JSON-serializable values."""
    if depth > 10:
        return str(value)
    if value is None or isinstance(value, (str, bool, int)):
        return value
    if isinstance(value, float):
        return value if math.isfinite(value) else str(value)
    if isinstance(value, bytes):
        return value.decode("utf-8", "replace")
    if isinstance(value, dict):
        return {str(k): json_safe(v, depth + 1) for k, v in value.items()}
    if isinstance(value, (list, tuple)):
        return [json_safe(v, depth + 1) for v in value]
    try:
        if hasattr(value, "__iter__"):
            return [json_safe(v, depth + 1) for v in value]
    except Exception:
        pass
    try:
        return str(value)
    except Exception:
        return repr(value)


def safe_filename(value):
    """Windows/POSIX safe filename component; avoids fragile backslash escapes."""
    text = str(value).strip().replace("/", "_").replace(chr(92), "_")
    for bad in (":", "*", "?", '"', "<", ">", "|"):
        text = text.replace(bad, "_")
    text = text.strip(" .")
    return (text or "geometry_export")[:120]

def get_closed(prim):

def distance3(a, b):
    dx = float(a[0]) - float(b[0])
    dy = float(a[1]) - float(b[1])
    dz = float(a[2]) - float(b[2])
    return math.sqrt(dx * dx + dy * dy + dz * dz)


def curve_length(pos_list, closed):
    if len(pos_list) < 2:
        return 0.0
    total = 0.0
    for i in range(len(pos_list) - 1):
        total += distance3(pos_list[i], pos_list[i + 1])
    if closed and len(pos_list) > 2:
        total += distance3(pos_list[-1], pos_list[0])
    return float(total)


def xz_curve_length(pos_list, closed):
    if len(pos_list) < 2:
        return 0.0
    total = 0.0
    edge_count = len(pos_list) if closed else len(pos_list) - 1
    for i in range(edge_count):
        a = pos_list[i]
        b = pos_list[(i + 1) % len(pos_list)]
        dx = float(b[0]) - float(a[0])
        dz = float(b[2]) - float(a[2])
        total += math.sqrt(dx * dx + dz * dz)
    return float(total)


def bbox_from_positions(positions):
    if not positions:
        return None
    return {
        "min": [min(float(p[k]) for p in positions) for k in range(3)],
        "max": [max(float(p[k]) for p in positions) for k in range(3)],
    }


def attribute_schema(attribs):
    result = []
    for attrib in attribs:
        row = {"name": attrib.name()}
        for key, getter in (
            ("type", lambda a: a.type()),
            ("data_type", lambda a: a.dataType()),
            ("size", lambda a: a.size()),
            ("is_array", lambda a: a.isArrayType()),
        ):
            try:
                row[key] = json_safe(getter(attrib))
            except Exception:
                pass
        result.append(row)
    return result

def clean_consecutive_positions(prim):
    pos = []
    for vertex in prim.vertices():
        p = vertex.point().position()
        q = (float(p[0]), float(p[1]), float(p[2]))
        if not pos or distance3(q, pos[-1]) > 1e-9:
            pos.append(q)
    closed = get_closed(prim)
    if closed and len(pos) >= 3 and distance3(pos[0], pos[-1]) <= 1e-9:
        pos.pop()
    return pos, closed


def map_v16_stroke_ids(curve_geo, primitive_rows):
    """Recreate V16's sequential sid assignment for valid XZ curves.

    In V16, stroke_id/sid is an index among valid strokes after consecutive duplicate
    vertices are removed and curves whose projected XZ length is <= 1e-6 are skipped.
    This mapping is used only to connect cover_stroke_ids to source Primitive numbers.
    """
    sid_to_prim = {}
    prim_to_sid = {}
    skipped = []
    sid = 0
    for prim in curve_geo.prims():
        pos, closed = clean_consecutive_positions(prim)
        if len(pos) < 2:
            skipped.append(int(prim.number()))
            continue
        length_xz = xz_curve_length(pos, closed)
        if length_xz <= 1e-6:
            skipped.append(int(prim.number()))
            continue
        prim_num = int(prim.number())
        sid_to_prim[sid] = prim_num
        prim_to_sid[prim_num] = sid
        sid += 1
    return sid_to_prim, prim_to_sid, skipped


def parse_id_list(value):
    if value is None:
        return []
    if isinstance(value, (int, float)):
        return [int(value)]
    if isinstance(value, (list, tuple)):
        out = []
        for item in value:
            try:
                out.append(int(item))
            except Exception:
                pass
        return out
    text = str(value).strip()
    if not text:
        return []
    out = []
    for token in text.replace(",", ";").split(";"):
        token = token.strip()
        if not token:
            continue
        try:
            out.append(int(token))
        except Exception:
            continue
    return out

def spatial_pairs(points, radius, plane="3d", max_rows=200000):
    """Generate near-point diagnostic rows with a grid index; never all-pairs O(N^2)."""
    radius = float(radius)
    if radius <= 0 or len(points) < 2:
        return [], False
    cell = radius
    grid = defaultdict(list)
    rows = []
    truncated = False
    r2 = radius * radius
    for idx, row in enumerate(points):
        x, y, z = row["position"]
        if plane == "xz":
            key = (int(math.floor(x / cell)), int(math.floor(z / cell)))
            neighbor_offsets = ((dx, dz) for dx in (-1, 0, 1) for dz in (-1, 0, 1))
            for dx, dz in neighbor_offsets:
                for j in grid.get((key[0] + dx, key[1] + dz), []):
                    ox, oy, oz = points[j]["position"]
                    dxv, dzv = x - ox, z - oz
                    d2 = dxv * dxv + dzv * dzv
                    if d2 <= r2:
                        if len(rows) >= max_rows:
                            truncated = True
                            return rows, truncated
                        attrs_a = points[j].get("attributes", {})
                        attrs_b = row.get("attributes", {})
                        rows.append({
                            "point_a": points[j]["point_number"], "point_b": row["point_number"],
                            "distance_xz": math.sqrt(d2), "distance_3d": distance3((ox, oy, oz), (x, y, z)),
                            "a_x": ox, "a_y": oy, "a_z": oz, "b_x": x, "b_y": y, "b_z": z,
                            "a_depth_layer": attrs_a.get("depth_layer", ""),
                            "b_depth_layer": attrs_b.get("depth_layer", ""),
                            "a_stroke_id": attrs_a.get("stroke_id", ""), "b_stroke_id": attrs_b.get("stroke_id", ""),
                            "a_role": attrs_a.get("point_role", ""), "b_role": attrs_b.get("point_role", ""),
                        })
            grid[key].append(idx)
        else:
            key = (int(math.floor(x / cell)), int(math.floor(y / cell)), int(math.floor(z / cell)))
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    for dz in (-1, 0, 1):
                        for j in grid.get((key[0] + dx, key[1] + dy, key[2] + dz), []):
                            ox, oy, oz = points[j]["position"]
                            dxv, dyv, dzv = x - ox, y - oy, z - oz
                            d2 = dxv * dxv + dyv * dyv + dzv * dzv
                            if d2 <= r2:
                                if len(rows) >= max_rows:
                                    truncated = True
                                    return rows, truncated
                                attrs_a = points[j].get("attributes", {})
                                attrs_b = row.get("attributes", {})
                                rows.append({
                                    "point_a": points[j]["point_number"], "point_b": row["point_number"],
                                    "distance_3d": math.sqrt(d2), "distance_xz": math.sqrt((x-ox)**2 + (z-oz)**2),
                                    "a_x": ox, "a_y": oy, "a_z": oz, "b_x": x, "b_y": y, "b_z": z,
                                    "a_depth_layer": attrs_a.get("depth_layer", ""),
                                    "b_depth_layer": attrs_b.get("depth_layer", ""),
                                    "a_stroke_id": attrs_a.get("stroke_id", ""), "b_stroke_id": attrs_b.get("stroke_id", ""),
                                    "a_role": attrs_a.get("point_role", ""), "b_role": attrs_b.get("point_role", ""),
                                })
            grid[key].append(idx)
    return rows, truncated


def write_text(path, text):
    target = path + ".tmp" if WRITE_ATOMICALLY else path
    with open(target, "w", encoding="utf-8", newline="") as f:
        f.write(text)
    if WRITE_ATOMICALLY:
        os.replace(target, path)


def write_json(path, payload):
    write_text(path, json.dumps(payload, ensure_ascii=False, indent=2, allow_nan=False))


def write_csv(path, headers, rows):
    target = path + ".tmp" if WRITE_ATOMICALLY else path
    with open(target, "w", newline="", encoding="utf-8-sig") as f:
        writer = csv.DictWriter(f, fieldnames=headers, extrasaction="ignore")
        writer.writeheader()
        for row in rows:
            writer.writerow(row)
    if WRITE_ATOMICALLY:
        os.replace(target, path)


def point_csv_rows(points):
    for row in points:
        pos = row["position"]
        yield {
            "point_number": row["point_number"],
            "x": pos[0], "y": pos[1], "z": pos[2],
            "attributes_json": json.dumps(row.get("attributes", {}), ensure_ascii=False)[:CSV_ATTRIBUTE_JSON_MAX_CHARS],
        }


def main_export():
    started = time.perf_counter()
    node = hou.pwd()
    inputs = list(node.inputs())

    # The node's first input is used for output passthrough. Both source input geometries are exported separately.
    max_index = max(int(CURVES_INPUT_INDEX), int(PLACEMENT_POINTS_INPUT_INDEX))
    if len(inputs) <= max_index:
        raise hou.NodeError(
            "python_sop_export_v16_dual: 需要两个已连接输入。输入0连接 python_stroke_analysis_v16（曲线），"
            "输入1连接 python_global_spacing_v16（布点）。当前 node.inputs() 数量={}。".format(len(inputs))
        )
    if inputs[CURVES_INPUT_INDEX] is None:
        raise hou.NodeError("python_sop_export_v16_dual: 输入0为空；请将 python_stroke_analysis_v16 连接到第一个输入。")
    if inputs[PLACEMENT_POINTS_INPUT_INDEX] is None:
        raise hou.NodeError("python_sop_export_v16_dual: 输入1为空；请将 python_global_spacing_v16 连接到第二个输入。")

    curve_node = inputs[CURVES_INPUT_INDEX]
    point_node = inputs[PLACEMENT_POINTS_INPUT_INDEX]
    curve_geo = curve_node.geometry()
    point_geo = point_node.geometry()

    out_geo = node.geometry()
    out_geo.clear()
    if PASSTHROUGH_CURVE_INPUT:
        out_geo.merge(curve_geo)

    if not EXPORT_ON_COOK:
        print("python_sop_export_v16_dual: EXPORT_ON_COOK=False; geometry passthrough only.")
        return

    try:
        export_dir = hou.expandString(EXPORT_DIR)
        export_dir = os.path.abspath(os.path.expanduser(export_dir))
        os.makedirs(export_dir, exist_ok=True)
    except Exception as exc:
        raise hou.NodeError("python_sop_export_v16_dual: 无法创建导出目录 {!r}: {}".format(EXPORT_DIR, exc))
    prefix = safe_filename(EXPORT_PREFIX)
    base = os.path.join(export_dir, prefix)

    curve_data = collect_geometry(curve_geo, curve_node, "stroke_analysis_curves")
    point_data = collect_geometry(point_geo, point_node, "global_spacing_points")

    sid_to_prim, prim_to_sid, skipped_for_sid = map_v16_stroke_ids(curve_geo, curve_data["primitives"])
    links, unmatched_primary, unresolved_coverage = build_point_curve_links(point_data, curve_data, sid_to_prim)

    point_rows_3d, truncated_3d = ([], False)
    point_rows_xz, truncated_xz = ([], False)
    if WRITE_NEAR_POINT_DIAGNOSTICS:
        point_rows_3d, truncated_3d = spatial_pairs(point_data["points"], NEAR_3D_RADIUS, "3d", int(MAX_PAIR_ROWS_PER_TABLE))
        point_rows_xz, truncated_xz = spatial_pairs(point_data["points"], NEAR_XZ_RADIUS, "xz", int(MAX_PAIR_ROWS_PER_TABLE))

    try:
        houdini_version = hou.applicationVersionString()
    except Exception:
        houdini_version = "unknown"

    point_y_values = sorted(set(round(float(p["position"][1]), 6) for p in point_data["points"]))
    point_attrs_all = [p.get("attributes", {}) for p in point_data["points"]]
    depth_counts = defaultdict(int)
    for attrs in point_attrs_all:
        try:
            depth_counts[str(int(attrs.get("depth_layer", 0)))] += 1
        except Exception:
            pass

    point_node_input_paths = []
    try:
        for upstream in point_node.inputs():
            point_node_input_paths.append(upstream.path() if upstream is not None else None)
    except Exception:
        point_node_input_paths = []
    curve_node_input_paths = []
    try:
        for upstream in curve_node.inputs():
            curve_node_input_paths.append(upstream.path() if upstream is not None else None)
    except Exception:
        curve_node_input_paths = []
    spacing_input0_matches_curve_input = bool(point_node_input_paths and point_node_input_paths[0] == curve_node.path())

    summary = {
        "schema_version": "2.0-dual-input",
        "exporter_node_path": node.path(),
        "export_time_local": time.strftime("%Y-%m-%d %H:%M:%S"),
        "houdini_version": houdini_version,
        "input_order": {
            str(CURVES_INPUT_INDEX): {"role": "curves", "node_path": curve_node.path(), "upstream_paths": curve_node_input_paths},
            str(PLACEMENT_POINTS_INPUT_INDEX): {"role": "placement_points", "node_path": point_node.path(), "upstream_paths": point_node_input_paths},
        },
        "source_chain_check": {
            "global_spacing_input0_is_stroke_analysis_node": spacing_input0_matches_curve_input,
            "note": "For exact primitive mapping, connect python_stroke_analysis_v16 directly to input0 of python_global_spacing_v16, then also connect that same stroke-analysis node to export input0.",
        },
        "coordinate_note": "Coordinates are SOP-local for each input. Parent object world transform matrices are included separately. Do not compare world positions without applying these transforms if the two source objects differ.",
        "curves": curve_data["summary"],
        "placement_points": point_data["summary"],
        "matching": {
            "mapping_rule_primary": "placement point source_prim -> curves primitive_number (exact ID match when source geometry order is preserved)",
            "mapping_rule_coverage": "cover_stroke_ids/stroke_id -> V16 sequential valid XZ stroke ID map inferred from input 0 curves",
            "curve_primitives_valid_for_stroke_ids": len(sid_to_prim),
            "curve_primitives_skipped_when_rebuilding_stroke_ids": skipped_for_sid,
            "placement_point_rows": len(point_data["points"]),
            "point_stroke_link_rows": len(links),
            "unmatched_source_prim_point_count": len(unmatched_primary),
            "unresolved_coverage_id_count": len(unresolved_coverage),
            "unmatched_source_prim_point_numbers": unmatched_primary[:1000],
            "unresolved_coverage_examples": unresolved_coverage[:1000],
        },
        "placement_depth_diagnostics": {
            "unique_y_values_rounded_1e-6": point_y_values,
            "depth_layer_counts": dict(depth_counts),
            "point_count": len(point_data["points"]),
        },
        "near_point_diagnostics": {
            "enabled": bool(WRITE_NEAR_POINT_DIAGNOSTICS),
            "3d_radius": float(NEAR_3D_RADIUS),
            "3d_pair_count": len(point_rows_3d),
            "3d_truncated": bool(truncated_3d),
            "xz_radius": float(NEAR_XZ_RADIUS),
            "xz_pair_count": len(point_rows_xz),
            "xz_truncated": bool(truncated_xz),
        },
        "export_dir": export_dir,
        "export_prefix": prefix,
    }

    combined = {
        "summary": summary,
        "stroke_geometry": {k: v for k, v in curve_data.items() if not k.startswith("_")},
        "placement_point_geometry": {k: v for k, v in point_data.items() if not k.startswith("_")},
        "point_to_stroke_links": links,
    }

    written = []
    csv_field_cap = max(500, int(CSV_ATTRIBUTE_JSON_MAX_CHARS))
    if WRITE_COMBINED_JSON:
        path = base + "_combined.json"
        write_json(path, combined)
        written.append(path)

    if WRITE_CSV_TABLES:
        path = base + "_curve_points.csv"
        write_csv(path, ["point_number", "x", "y", "z", "attributes_json"], point_csv_rows(curve_data["points"]))
        written.append(path)

        path = base + "_curve_primitives.csv"
        write_csv(path, ["primitive_number", "primitive_type", "is_closed", "vertex_count", "curve_length_3d", "curve_length_xz", "point_numbers_json", "positions_json", "attributes_json"], (
            {"primitive_number": r["primitive_number"], "primitive_type": r["primitive_type"], "is_closed": r["is_closed"],
             "vertex_count": r["vertex_count"], "curve_length_3d": r["curve_length_3d"], "curve_length_xz": r["curve_length_xz"],
             "point_numbers_json": json.dumps(r["point_numbers"], ensure_ascii=False),
             "positions_json": json.dumps(r["positions"], ensure_ascii=False),
             "attributes_json": json.dumps(r["attributes"], ensure_ascii=False)[:csv_field_cap]}
            for r in curve_data["primitives"]))
        written.append(path)

        path = base + "_curve_vertices.csv"
        write_csv(path, ["primitive_number", "vertex_order", "point_number", "x", "y", "z", "cumulative_arc_length_3d", "attributes_json"], (
            {"primitive_number": r["primitive_number"], "vertex_order": r["vertex_order"], "point_number": r["point_number"],
             "x": r["x"], "y": r["y"], "z": r["z"], "cumulative_arc_length_3d": r["cumulative_arc_length_3d"],
             "attributes_json": json.dumps(r["attributes"], ensure_ascii=False)[:csv_field_cap]}
            for r in curve_data["vertices"]))
        written.append(path)

        path = base + "_curve_segments.csv"
        write_csv(path, ["primitive_number", "segment_index", "point_a", "point_b", "ax", "ay", "az", "bx", "by", "bz", "length_3d", "length_xz"], curve_data["segments"])
        written.append(path)

        path = base + "_placement_points.csv"
        placement_rows = []
        for r in point_data["points"]:
            attrs = r.get("attributes", {})
            pos = r["position"]
            placement_rows.append({
                "point_number": r["point_number"], "x": pos[0], "y": pos[1], "z": pos[2],
                "global_point_id": attrs.get("global_point_id", ""),
                "source_prim": attrs.get("source_prim", ""), "stroke_id": attrs.get("stroke_id", ""),
                "continuity_chain_id": attrs.get("continuity_chain_id", ""),
                "arc_length": attrs.get("arc_length", ""), "ideal_arc_length": attrs.get("ideal_arc_length", ""),
                "point_role": attrs.get("point_role", ""), "point_priority": attrs.get("point_priority", ""),
                "is_endpoint": attrs.get("is_endpoint", ""), "is_corner": attrs.get("is_corner", ""),
                "is_junction": attrs.get("is_junction", ""), "depth_layer": attrs.get("depth_layer", ""),
                "depth_offset": attrs.get("depth_offset", ""), "cover_stroke_ids": attrs.get("cover_stroke_ids", ""),
                "attributes_json": json.dumps(attrs, ensure_ascii=False)[:csv_field_cap],
            })
        write_csv(path, ["point_number", "x", "y", "z", "global_point_id", "source_prim", "stroke_id", "continuity_chain_id", "arc_length", "ideal_arc_length", "point_role", "point_priority", "is_endpoint", "is_corner", "is_junction", "depth_layer", "depth_offset", "cover_stroke_ids", "attributes_json"], placement_rows)
        written.append(path)

        path = base + "_point_to_stroke_links.csv"
        link_headers = ["placement_point_number", "global_point_id", "source_prim", "source_stroke_id", "cover_stroke_id", "matched_curve_primitive", "match_status", "is_primary_source_curve", "stroke_kind", "orientation_deg", "stroke_length_xz", "shape_group_id", "related_prim_ids", "arc_length", "ideal_arc_length", "point_role", "depth_layer", "depth_offset"]
        write_csv(path, link_headers, links)
        written.append(path)

    if WRITE_NEAR_POINT_DIAGNOSTICS:
        path = base + "_near_pairs_3d.csv"
        write_csv(path, ["point_a", "point_b", "distance_3d", "distance_xz", "a_x", "a_y", "a_z", "b_x", "b_y", "b_z", "a_depth_layer", "b_depth_layer", "a_stroke_id", "b_stroke_id", "a_role", "b_role"], point_rows_3d)
        written.append(path)
        path = base + "_near_pairs_front_xz.csv"
        write_csv(path, ["point_a", "point_b", "distance_xz", "distance_3d", "a_x", "a_y", "a_z", "b_x", "b_y", "b_z", "a_depth_layer", "b_depth_layer", "a_stroke_id", "b_stroke_id", "a_role", "b_role"], point_rows_xz)
        written.append(path)

    summary_path = base + "_summary.json"
    write_json(summary_path, summary)
    written.append(summary_path)

    # README bundled beside the data for easier downstream inspection.
    readme_path = base + "_README.txt"
    readme = (
        "Houdini 20.5 dual-input linework + placement point export\n"
        "Input 0 (curves): {0}\n"
        "Input 1 (placement points): {1}\n"
        "Exporter: {2}\n"
        "The combined JSON contains both source geometries and point-to-stroke links.\n"
        "Coordinates are SOP-local for each source; see summary transforms before comparing object-space positions.\n"
        "Primary mapping uses placement point source_prim -> curve primitive_number. cover_stroke_ids are mapped through the inferred V16 sequential stroke id map.\n"
        "Files: {3}\n"
    ).format(curve_node.path(), point_node.path(), node.path(), "\n       ".join(os.path.basename(p) for p in written))
    write_text(readme_path, readme)
    written.append(readme_path)

    zip_path = base + "_bundle.zip"
    if WRITE_ZIP_BUNDLE:
        zip_tmp = zip_path + ".tmp"
        with zipfile.ZipFile(zip_tmp, "w", compression=zipfile.ZIP_DEFLATED, compresslevel=6) as zf:
            for path in written:
                if os.path.isfile(path):
                    zf.write(path, arcname=os.path.basename(path))
        if os.path.exists(zip_path):
            os.remove(zip_path)
        os.replace(zip_tmp, zip_path)

    elapsed = time.perf_counter() - started
    print("=" * 76)
    print("python_sop_export_v16_dual: export completed")
    print("Input 0 curves :", curve_node.path())
    print("Input 1 points:", point_node.path())
    print("Curve prims / vertices / segments:", len(curve_data["primitives"]), "/", len(curve_data["vertices"]), "/", len(curve_data["segments"]))
    print("Placement points:", len(point_data["points"]))
    print("Point-to-stroke links:", len(links))
    print("Unmatched source_prim points:", len(unmatched_primary))
    print("Unresolved cover_stroke_ids:", len(unresolved_coverage))
    print("3D near pairs / truncated:", len(point_rows_3d), "/", truncated_3d)
    print("XZ near pairs / truncated:", len(point_rows_xz), "/", truncated_xz)
    print("Output directory:", export_dir)
    print("ZIP bundle:", zip_path if WRITE_ZIP_BUNDLE else "disabled")
    print("Export seconds:", round(elapsed, 4))
    print("Files written:")
    for path in written:
        print("  ", path)
    print("=" * 76)


main_export()
